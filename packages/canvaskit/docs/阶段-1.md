`src/engine/Engine.ts`

```ts
import CanvasKitInit from 'canvaskit-wasm';
import type * as CanvasKitModule from 'canvaskit-wasm';
import { EngineOptions } from './Config';
import { RenderContext } from '../renderer/RenderContext';
import { Renderer } from '../renderer/Renderer';
import { Ticker } from '../renderer/Ticker'; // small ticker implemented below or in animation module
import { Group } from '../scene/Node';

export class Engine {
  options: EngineOptions;
  canvasKit!: typeof CanvasKitModule;
  context!: RenderContext;
  renderer!: Renderer;
  root: Group;
  private ticker!: Ticker;

  constructor(options: EngineOptions) {
    this.options = { dpr: window.devicePixelRatio || 1, ...options };
    this.root = new Group();
  }

  async init() {
    // load CanvasKit - 默认会从 node_modules canvaskit-wasm 提供的路径加载 wasm
    this.canvasKit = await CanvasKitInit({
      locateFile: (file: string) => {
        // 你可以自定义 public 路径
        return `/node_modules/canvaskit-wasm/bin/${file}`;
      }
    });
    this.context = new RenderContext(this.canvasKit, this.options.canvas, this.options.dpr!);
    this.renderer = new Renderer(this.context);
    this.ticker = new Ticker();
    // 简单绑定每帧渲染
    this.ticker.onTick = () => this.render();
  }

  start() {
    this.ticker.start();
  }

  render() {
    this.renderer.render(this.root);
  }

  dispose() {
    this.context.dispose();
  }
}
```

> 注：我在 `Engine` 中使用 `canvaskit-wasm` 包的默认加载方式；你也可以改为 CDN 或本地 public 下的 wasm。

`src/renderer/RenderContext.ts`

```ts
export class RenderContext {
  CanvasKit: any;
  canvas: HTMLCanvasElement;
  dpr: number;
  surface: any; // CanvasKit.Surface

  constructor(CanvasKit: any, canvas: HTMLCanvasElement, dpr = 1) {
    this.CanvasKit = CanvasKit;
    this.canvas = canvas;
    this.dpr = dpr;
    this.createSurface();
  }

  createSurface() {
    const ck = this.CanvasKit;
    // adjust canvas size for DPR
    const logicalW = this.canvas.clientWidth || this.canvas.width;
    const logicalH = this.canvas.clientHeight || this.canvas.height;
    this.canvas.width = Math.round(logicalW * this.dpr);
    this.canvas.height = Math.round(logicalH * this.dpr);
    this.surface = ck.MakeCanvasSurface(this.canvas);
    if (!this.surface) throw new Error('Failed to create CanvasKit surface');
  }

  getCanvas() {
    return this.surface.getCanvas();
  }

  flush() {
    this.surface.flush();
  }

  clear() {
    const ck = this.CanvasKit;
    const canvas = this.getCanvas();
    canvas.clear(ck.Color(0, 0, 0, 0));
  }

  dispose() {
    try {
      this.surface?.dispose?.();
    } catch (e) {}
  }
}
```

`src/renderer/Renderer.ts`

```ts
import { RenderContext } from './RenderContext';
import { Node } from '../scene/Node';

export class Renderer {
  ctx: RenderContext;

  constructor(ctx: RenderContext) {
    this.ctx = ctx;
  }

  render(root: Node) {
    const canvas = this.ctx.getCanvas();
    // clear
    this.ctx.clear();
    canvas.save();
    // scale to devicePixelRatio
    canvas.scale(this.ctx.dpr, this.ctx.dpr);
    this.drawNode(canvas, root);
    canvas.restore();
    this.ctx.flush();
  }

  private drawNode(canvas: any, node: Node) {
    if (!node.visible) return;
    canvas.save();
    // apply node transform (matrix as [a,b,c,d,e,f])
    const m = node.worldMatrix.array;
    // CanvasKit expects 3x3 2D affine matrix - use concat or setMatrix if available
    canvas.concat(m);
    node.onDraw(canvas);
    for (const child of node.children) {
      this.drawNode(canvas, child);
    }
    canvas.restore();
  }
}
```

`src/core/Matrix2D.ts`

```ts
// 2D affine represented as Float32Array [a, b, c, d, e, f]
// corresponds to matrix:
// [ a c e ]
// [ b d f ]
// [ 0 0 1 ]
export class Matrix2D {
  array: Float32Array;
  constructor(a = 1, b = 0, c = 0, d = 1, e = 0, f = 0) {
    this.array = new Float32Array([a, b, c, d, e, f]);
  }

  static identity() { return new Matrix2D(); }

  multiply(other: Matrix2D) {
    const a = this.array, b = other.array;
    const a0 = a[0], a1 = a[1], a2 = a[2], a3 = a[3], a4 = a[4], a5 = a[5];
    const b0 = b[0], b1 = b[1], b2 = b[2], b3 = b[3], b4 = b[4], b5 = b[5];
    const r0 = a0 * b0 + a2 * b1;
    const r1 = a1 * b0 + a3 * b1;
    const r2 = a0 * b2 + a2 * b3;
    const r3 = a1 * b2 + a3 * b3;
    const r4 = a0 * b4 + a2 * b5 + a4;
    const r5 = a1 * b4 + a3 * b5 + a5;
    this.array[0] = r0; this.array[1] = r1; this.array[2] = r2;
    this.array[3] = r3; this.array[4] = r4; this.array[5] = r5;
    return this;
  }

  setTranslate(x: number, y: number) {
    this.array[4] = x; this.array[5] = y; return this;
  }

  clone() {
    const a = this.array;
    return new Matrix2D(a[0], a[1], a[2], a[3], a[4], a[5]);
  }
}
```

`src/scene/Node.ts`

```ts
import { Matrix2D } from '../core/Matrix2D';

export class Node {
  parent: Node | null = null;
  children: Node[] = [];
  visible = true;
  zIndex = 0;
  localMatrix = Matrix2D.identity();
  worldMatrix = Matrix2D.identity();
  worldDirty = true;

  constructor() {}

  add(child: Node) {
    child.parent = this;
    this.children.push(child);
    this.markDirty();
  }

  remove(child: Node) {
    const idx = this.children.indexOf(child);
    if (idx >= 0) {
      this.children.splice(idx, 1);
      child.parent = null;
    }
  }

  markDirty() {
    this.worldDirty = true;
    for (const c of this.children) c.markDirty();
  }

  updateWorldMatrix() {
    if (!this.worldDirty) return;
    if (this.parent) {
      this.worldMatrix = this.parent.worldMatrix.clone().multiply(this.localMatrix);
    } else {
      this.worldMatrix = this.localMatrix.clone();
    }
    this.worldDirty = false;
    for (const c of this.children) c.updateWorldMatrix();
  }

  // override in subclasses
  onDraw(canvas: any) {}

  setTransform(a:number,b:number,c:number,d:number,e:number,f:number) {
    this.localMatrix = new Matrix2D(a,b,c,d,e,f);
    this.markDirty();
  }
}

export class Group extends Node {}
```

`src/scene/PathBuilder.ts`

```ts
// Simple command list path builder to allow lazy CanvasKit.Path rebuild.
export type PathCommand =
  | { cmd: 'moveTo'; x:number; y:number }
  | { cmd: 'lineTo'; x:number; y:number }
  | { cmd: 'arc'; x:number; y:number; r:number } // simplified
  | { cmd: 'close' };

export class PathBuilder {
  commands: PathCommand[] = [];
  version = 0;
  add(cmd: PathCommand) { this.commands.push(cmd); this.version++; }
  clear() { this.commands.length = 0; this.version++; }

  build(CanvasKit: any) {
    const p = new CanvasKit.Path();
    for (const c of this.commands) {
      if (c.cmd === 'moveTo') p.moveTo(c.x, c.y);
      else if (c.cmd === 'lineTo') p.lineTo(c.x, c.y);
      else if (c.cmd === 'arc') p.addCircle(c.x, c.y, c.r, CanvasKit.PathDirection.CW);
      else if (c.cmd === 'close') p.close();
    }
    return p;
  }
}
```

`src/scene/Shape.ts`

```ts
import { Node } from './Node';
import { PathBuilder } from './PathBuilder';

export class Shape extends Node {
  pathBuilder: PathBuilder;
  private _ckPath: any | null = null;
  private _builtVersion = -1;
  fillStyle = '#ff0000'; // simple hex string for demo
  strokeStyle: string | null = null;

  constructor(pathBuilder?: PathBuilder) {
    super();
    this.pathBuilder = pathBuilder ?? new PathBuilder();
  }

  getPath(CanvasKit: any) {
    if (!this._ckPath || this._builtVersion !== this.pathBuilder.version) {
      this._ckPath?.delete?.();
      this._ckPath = this.pathBuilder.build(CanvasKit);
      this._builtVersion = this.pathBuilder.version;
    }
    return this._ckPath;
  }

  onDraw(canvas: any) {
    // CanvasKit instance is accessible via canvas.getTotalMatrix()... but we pass CanvasKit in render step instead.
    // Here we assume caller will bind CanvasKit to canvas via closure or Renderer has access.
    // For demo, use a simple approach: draw red fill via provided path already constructed with CanvasKit
    const CanvasKit = (canvas as any).__CanvasKit;
    if (!CanvasKit) return;
    const path = this.getPath(CanvasKit);
    // Fill
    const paint = new CanvasKit.Paint();
    paint.setAntiAlias(true);
    const c = this.parseColor(this.fillStyle);
    paint.setColor(CanvasKit.Color(c.r, c.g, c.b, c.a));
    paint.setStyle(CanvasKit.PaintStyle.Fill);
    canvas.drawPath(path, paint);
    paint.delete();
  }

  private parseColor(hex: string) {
    // very simple hex #rrggbb
    const r = parseInt(hex.slice(1,3),16);
    const g = parseInt(hex.slice(3,5),16);
    const b = parseInt(hex.slice(5,7),16);
    return { r, g, b, a: 1 };
  }

  dispose() {
    this._ckPath?.delete?.();
    this._ckPath = null;
  }
}
```

> 说明：上面 `Shape.onDraw` 中临时创建 Paint 是简化示例。**生产里应使用 PaintPool**（后续阶段会实现），这里为了让示例一步能跑通先这样做。

`src/renderer/Ticker.ts` (简单实现)

```ts
export class Ticker {
  running = false;
  last = 0;
  onTick: ((dt:number)=>void) | null = null;

  start() {
    if (this.running) return;
    this.running = true;
    this.last = performance.now();
    const loop = (t: number) => {
      if (!this.running) return;
      const dt = t - this.last;
      this.last = t;
      this.onTick?.(dt);
      requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
  }

  stop() { this.running = false; }
}
```

# 四、运行注意与下一步建议

1. **CanvasKit wasm 路径**：`canvaskit-wasm` 包会在 `node_modules/canvaskit-wasm/bin` 放 wasm 文件；Vite dev 环境通常会直接加载。如果出现 404，请把 `node_modules/canvaskit-wasm/bin/` 中的 `canvaskit.wasm` 复制到 `public/` 并在 `Engine` 的 `locateFile` 返回 `/public/canvaskit.wasm` 路径。

2. **内存管理**：示例中为简洁直接在 `Shape.onDraw` 创建 `Paint` 并立即 `delete()`，实际生产要使用 `PaintPool` 避免频繁 new/delete。也请注意 Path 对象要 `delete()` 或用对象池复用（我在 Shape 中提供了 `dispose()` 示例）。

3. **Render 矩阵**：`Renderer` 使用 `canvas.concat(m)`，这里假定 CanvasKit 能接受 6 元数组。如果兼容性问题，可改为 `canvas.concat` 接受 `Float32Array` 或 `canvas.setMatrix` 等 API（请查阅当前 CanvasKit 版本的 API）。
