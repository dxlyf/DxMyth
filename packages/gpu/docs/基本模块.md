我将为你设计一个现代的、同时支持WebGL和WebGPU的声明式图形库，采用类REGL的设计模式。

## 1. 核心架构设计

```javascript
// lib/gfx-core.js
class GFX {
    constructor(canvas, options = {}) {
        this.canvas = canvas;
        this.options = {
            contextType: 'auto', // 'webgl', 'webgl2', 'webgpu', 'auto'
            alpha: true,
            antialias: true,
            powerPreference: 'high-performance',
            ...options
        };
        
        this.context = null;
        this.rendererType = null;
        this.extensions = {};
        this.capabilities = {};
        
        this._nextId = 1;
        this._resources = new Map();
        this._frameCallbacks = new Set();
        this._pendingFrames = new Set();
        
        this._stats = {
            drawCalls: 0,
            triangles: 0,
            textures: 0,
            buffers: 0,
            shaders: 0
        };
        
        this.init();
    }
    
    async init() {
        try {
            // 尝试WebGPU
            if (this.shouldUseWebGPU()) {
                this.context = await this.initWebGPU();
                this.rendererType = 'webgpu';
            } else {
                // 回退到WebGL
                this.context = this.initWebGL();
                this.rendererType = 'webgl';
            }
            
            this.detectCapabilities();
            this.setupDefaultState();
            
            console.log(`GFX initialized with ${this.rendererType.toUpperCase()}`);
            
        } catch (error) {
            console.error('Failed to initialize GFX:', error);
            throw error;
        }
    }
    
    shouldUseWebGPU() {
        const { contextType } = this.options;
        
        if (contextType === 'webgl') return false;
        if (contextType === 'webgpu') return true;
        
        // 自动检测：优先WebGPU
        return 'gpu' in navigator;
    }
    
    async initWebGPU() {
        if (!('gpu' in navigator)) {
            throw new Error('WebGPU not supported');
        }
        
        const adapter = await navigator.gpu.requestAdapter({
            powerPreference: this.options.powerPreference
        });
        
        if (!adapter) {
            throw new Error('No WebGPU adapter found');
        }
        
        const device = await adapter.requestDevice();
        const context = this.canvas.getContext('webgpu');
        
        const presentationFormat = navigator.gpu.getPreferredCanvasFormat();
        
        context.configure({
            device,
            format: presentationFormat,
            alphaMode: this.options.alpha ? 'premultiplied' : 'opaque',
            usage: GPUTextureUsage.RENDER_ATTACHMENT,
            compositingAlphaMode: 'premultiplied'
        });
        
        return { device, context, adapter, format: presentationFormat };
    }
    
    initWebGL() {
        const gl = this.canvas.getContext(this.options.contextType === 'webgl2' ? 'webgl2' : 'webgl', {
            alpha: this.options.alpha,
            antialias: this.options.antialias,
            powerPreference: this.options.powerPreference,
            preserveDrawingBuffer: false,
            depth: true,
            stencil: true,
            premultipliedAlpha: true
        });
        
        if (!gl) {
            throw new Error('WebGL not supported');
        }
        
        return gl;
    }
    
    detectCapabilities() {
        if (this.rendererType === 'webgpu') {
            this.capabilities = {
                maxTextureSize: 8192,
                maxTextureUnits: 16,
                instancing: true,
                compute: true,
                storageBuffers: true,
                uniformBuffers: true,
                samplerLOD: true,
                anisotropy: 16,
                floatTextures: true,
                depthTexture: true,
                multisample: 4
            };
        } else {
            const gl = this.context;
            
            this.capabilities = {
                maxTextureSize: gl.getParameter(gl.MAX_TEXTURE_SIZE),
                maxTextureUnits: gl.getParameter(gl.MAX_TEXTURE_IMAGE_UNITS),
                instancing: !!gl.getExtension('ANGLE_instanced_arrays'),
                compute: !!gl.getExtension('WEBGL_2_compute') || !!gl.getExtension('WEBGL_compute'),
                storageBuffers: !!gl.getExtension('WEBGL_buffer_storage'),
                uniformBuffers: !!gl.getExtension('WEBGL_uniform_buffer_object'),
                samplerLOD: true,
                anisotropy: gl.getExtension('EXT_texture_filter_anisotropic')?.MAX_TEXTURE_MAX_ANISOTROPY_EXT || 1,
                floatTextures: !!gl.getExtension('OES_texture_float'),
                depthTexture: !!gl.getExtension('WEBGL_depth_texture'),
                multisample: gl.getParameter(gl.MAX_SAMPLES) || 1
            };
            
            // 存储扩展
            this.extensions.instancedArrays = gl.getExtension('ANGLE_instanced_arrays');
            this.extensions.anisotropic = gl.getExtension('EXT_texture_filter_anisotropic');
            this.extensions.depthTexture = gl.getExtension('WEBGL_depth_texture');
            this.extensions.textureFloat = gl.getExtension('OES_texture_float');
            this.extensions.drawBuffers = gl.getExtension('WEBGL_draw_buffers');
        }
    }
    
    setupDefaultState() {
        if (this.rendererType === 'webgpu') {
            // WebGPU状态设置
            const { device } = this.context;
            
            // 创建默认采样器
            this._defaultSampler = device.createSampler({
                magFilter: 'linear',
                minFilter: 'linear',
                mipmapFilter: 'linear',
                addressModeU: 'repeat',
                addressModeV: 'repeat'
            });
            
        } else {
            // WebGL状态设置
            const gl = this.context;
            
            gl.enable(gl.DEPTH_TEST);
            gl.depthFunc(gl.LEQUAL);
            gl.enable(gl.CULL_FACE);
            gl.cullFace(gl.BACK);
            gl.frontFace(gl.CCW);
            
            if (this.options.alpha) {
                gl.enable(gl.BLEND);
                gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA);
            }
        }
    }
    
    // 资源管理
    _createResource(type, spec) {
        const id = this._nextId++;
        const resource = { id, type, spec, refCount: 1 };
        this._resources.set(id, resource);
        return id;
    }
    
    _getResource(id) {
        return this._resources.get(id);
    }
    
    _destroyResource(id) {
        const resource = this._resources.get(id);
        if (resource && --resource.refCount <= 0) {
            this._cleanupResource(resource);
            this._resources.delete(id);
        }
    }
    
    _cleanupResource(resource) {
        const { type, spec } = resource;
        
        if (this.rendererType === 'webgpu') {
            const { device } = this.context;
            
            switch (type) {
                case 'buffer':
                    spec.buffer.destroy();
                    break;
                case 'texture':
                    spec.texture.destroy();
                    break;
                case 'sampler':
                    // GPU采样器不需要手动销毁
                    break;
                case 'shader':
                    spec.module.destroy();
                    break;
                case 'pipeline':
                    // 管道不需要手动销毁
                    break;
                case 'framebuffer':
                    // 帧缓冲区由GC管理
                    break;
            }
        } else {
            const gl = this.context;
            
            switch (type) {
                case 'buffer':
                    gl.deleteBuffer(spec.buffer);
                    break;
                case 'texture':
                    gl.deleteTexture(spec.texture);
                    break;
                case 'sampler':
                    // WebGL没有独立的采样器对象
                    break;
                case 'shader':
                    gl.deleteShader(spec.shader);
                    break;
                case 'program':
                    gl.deleteProgram(spec.program);
                    break;
                case 'framebuffer':
                    gl.deleteFramebuffer(spec.framebuffer);
                    break;
                case 'renderbuffer':
                    gl.deleteRenderbuffer(spec.renderbuffer);
                    break;
                case 'vertexArray':
                    gl.deleteVertexArray(spec.vao);
                    break;
            }
        }
    }
    
    // 渲染循环
    frame(callback) {
        const frameId = this._nextId++;
        const wrappedCallback = () => {
            try {
                callback();
            } finally {
                this._pendingFrames.delete(frameId);
            }
        };
        
        this._pendingFrames.add(frameId);
        this._frameCallbacks.add(wrappedCallback);
        
        if (this._pendingFrames.size === 1) {
            this._requestFrame();
        }
        
        return () => {
            this._pendingFrames.delete(frameId);
            this._frameCallbacks.delete(wrappedCallback);
        };
    }
    
    _requestFrame() {
        requestAnimationFrame(() => {
            this._stats.drawCalls = 0;
            this._stats.triangles = 0;
            
            this._frameCallbacks.forEach(callback => callback());
            
            if (this._pendingFrames.size > 0) {
                this._requestFrame();
            }
        });
    }
    
    // 公共API
    clear(color = [0, 0, 0, 1], depth = 1, stencil = 0) {
        if (this.rendererType === 'webgpu') {
            this.clearWebGPU(color, depth, stencil);
        } else {
            this.clearWebGL(color, depth, stencil);
        }
    }
    
    clearWebGPU(color, depth, stencil) {
        const { device, context } = this.context;
        const commandEncoder = device.createCommandEncoder();
        const textureView = context.getCurrentTexture().createView();
        
        const passEncoder = commandEncoder.beginRenderPass({
            colorAttachments: [{
                view: textureView,
                clearValue: { r: color[0], g: color[1], b: color[2], a: color[3] },
                loadOp: 'clear',
                storeOp: 'store'
            }],
            depthStencilAttachment: null
        });
        
        passEncoder.end();
        device.queue.submit([commandEncoder.finish()]);
    }
    
    clearWebGL(color, depth, stencil) {
        const gl = this.context;
        
        gl.clearColor(color[0], color[1], color[2], color[3]);
        gl.clearDepth(depth);
        gl.clearStencil(stencil);
        
        let clearMask = gl.COLOR_BUFFER_BIT;
        if (depth !== undefined) clearMask |= gl.DEPTH_BUFFER_BIT;
        if (stencil !== undefined) clearMask |= gl.STENCIL_BUFFER_BIT;
        
        gl.clear(clearMask);
    }
    
    getStats() {
        return { ...this._stats };
    }
    
    destroy() {
        // 清理所有资源
        this._resources.forEach(resource => {
            this._cleanupResource(resource);
        });
        this._resources.clear();
        
        this._frameCallbacks.clear();
        this._pendingFrames.clear();
        
        // 清理上下文
        if (this.rendererType === 'webgpu') {
            this.context.device.destroy();
        }
    }
}

// 工厂函数
function createGFX(canvas, options) {
    return new GFX(canvas, options);
}

export { GFX, createGFX };
```

```javascript
// lib/buffer.js
class BufferManager {
    constructor(gfx) {
        this.gfx = gfx;
        this.context = gfx.context;
        this.rendererType = gfx.rendererType;
    }
    
    create(spec) {
        const { data, usage, type = 'vertex', size, target } = spec;
        
        if (this.rendererType === 'webgpu') {
            return this.createWebGPUBuffer(data, usage, type, size);
        } else {
            return this.createWebGLBuffer(data, usage, type, target);
        }
    }
    
    createWebGPUBuffer(data, usage, type, size) {
        const { device } = this.context;
        
        let bufferSize = size;
        let mappedAtCreation = false;
        
        if (data) {
            bufferSize = data.byteLength || data.length * data.BYTES_PER_ELEMENT;
            mappedAtCreation = true;
        }
        
        const gpuUsage = this.mapBufferUsage(usage, type);
        
        const buffer = device.createBuffer({
            size: bufferSize,
            usage: gpuUsage,
            mappedAtCreation: mappedAtCreation
        });
        
        if (data && mappedAtCreation) {
            const arrayBuffer = buffer.getMappedRange();
            
            if (data instanceof ArrayBuffer) {
                new Uint8Array(arrayBuffer).set(new Uint8Array(data));
            } else {
                // TypedArray 或普通数组
                const view = new (data.constructor || Float32Array)(arrayBuffer);
                view.set(data);
            }
            
            buffer.unmap();
        }
        
        const id = this.gfx._createResource('buffer', { buffer, size: bufferSize });
        
        this.gfx._stats.buffers++;
        return id;
    }
    
    createWebGLBuffer(data, usage, type, target) {
        const gl = this.context;
        
        const glTarget = target || 
            (type === 'index' ? gl.ELEMENT_ARRAY_BUFFER : gl.ARRAY_BUFFER);
        
        const glUsage = this.mapBufferUsage(usage, type);
        
        const buffer = gl.createBuffer();
        gl.bindBuffer(glTarget, buffer);
        
        if (data) {
            gl.bufferData(glTarget, data, glUsage);
        } else if (size) {
            gl.bufferData(glTarget, size, glUsage);
        }
        
        gl.bindBuffer(glTarget, null);
        
        const id = this.gfx._createResource('buffer', { 
            buffer, 
            target: glTarget, 
            usage: glUsage 
        });
        
        this.gfx._stats.buffers++;
        return id;
    }
    
    mapBufferUsage(usage, type) {
        const usageMap = {
            'static': { webgl: this.context.STATIC_DRAW, webgpu: GPUBufferUsage.VERTEX | GPUBufferUsage.COPY_DST },
            'dynamic': { webgl: this.context.DYNAMIC_DRAW, webgpu: GPUBufferUsage.VERTEX | GPUBufferUsage.COPY_DST },
            'stream': { webgl: this.context.STREAM_DRAW, webgpu: GPUBufferUsage.VERTEX | GPUBufferUsage.COPY_DST },
            'storage': { webgl: this.context.DYNAMIC_READ, webgpu: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST },
            'uniform': { webgl: this.context.DYNAMIC_DRAW, webgpu: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST }
        };
        
        const baseUsage = usageMap[usage || 'static'];
        if (!baseUsage) {
            throw new Error(`Unknown buffer usage: ${usage}`);
        }
        
        if (this.rendererType === 'webgpu') {
            let gpuUsage = baseUsage.webgpu;
            
            if (type === 'index') {
                gpuUsage |= GPUBufferUsage.INDEX;
            }
            if (type === 'vertex') {
                gpuUsage |= GPUBufferUsage.VERTEX;
            }
            if (type === 'storage') {
                gpuUsage |= GPUBufferUsage.STORAGE;
            }
            
            return gpuUsage;
        } else {
            return baseUsage.webgl;
        }
    }
    
    update(bufferId, data, offset = 0) {
        const resource = this.gfx._getResource(bufferId);
        if (!resource || resource.type !== 'buffer') {
            throw new Error(`Buffer ${bufferId} not found`);
        }
        
        if (this.rendererType === 'webgpu') {
            const { device } = this.context;
            const { buffer } = resource.spec;
            
            device.queue.writeBuffer(
                buffer,
                offset,
                data.buffer || data,
                data.byteOffset || 0,
                data.byteLength || data.length * data.BYTES_PER_ELEMENT
            );
        } else {
            const gl = this.context;
            const { buffer, target } = resource.spec;
            
            gl.bindBuffer(target, buffer);
            gl.bufferSubData(target, offset, data);
            gl.bindBuffer(target, null);
        }
    }
    
    destroy(bufferId) {
        this.gfx._destroyResource(bufferId);
        this.gfx._stats.buffers--;
    }
}
```

```javascript
// lib/texture.js
class TextureManager {
    constructor(gfx) {
        this.gfx = gfx;
        this.context = gfx.context;
        this.rendererType = gfx.rendererType;
    }
    
    create(spec) {
        const {
            data,
            width,
            height,
            format = 'rgba8unorm',
            usage = 'sampled',
            mipmaps = true,
            wrap = 'repeat',
            filter = 'linear',
            anisotropy = 1
        } = spec;
        
        if (this.rendererType === 'webgpu') {
            return this.createWebGPUTexture(data, width, height, format, usage, mipmaps);
        } else {
            return this.createWebGLTexture(data, width, height, format, usage, mipmaps, wrap, filter, anisotropy);
        }
    }
    
    createWebGPUTexture(data, width, height, format, usage, mipmaps) {
        const { device } = this.context;
        
        const gpuUsage = this.mapTextureUsage(usage);
        const gpuFormat = this.mapTextureFormat(format);
        
        const texture = device.createTexture({
            size: { width, height },
            format: gpuFormat,
            usage: gpuUsage,
            mipLevelCount: mipmaps ? Math.floor(Math.log2(Math.max(width, height))) + 1 : 1
        });
        
        if (data) {
            this.uploadWebGPUTextureData(texture, data, width, height, gpuFormat);
        }
        
        const sampler = device.createSampler({
            magFilter: 'linear',
            minFilter: 'linear',
            mipmapFilter: 'linear',
            addressModeU: 'repeat',
            addressModeV: 'repeat'
        });
        
        const id = this.gfx._createResource('texture', { 
            texture, 
            sampler,
            format: gpuFormat,
            width,
            height
        });
        
        this.gfx._stats.textures++;
        return id;
    }
    
    createWebGLTexture(data, width, height, format, usage, mipmaps, wrap, filter, anisotropy) {
        const gl = this.context;
        
        const texture = gl.createTexture();
        gl.bindTexture(gl.TEXTURE_2D, texture);
        
        const glFormat = this.mapTextureFormat(format);
        
        // 设置纹理参数
        const glWrap = this.mapTextureWrap(wrap);
        const glFilter = this.mapTextureFilter(filter);
        
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, glWrap);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, glWrap);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, glFilter);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, 
            filter === 'nearest' ? gl.NEAREST : gl.LINEAR);
        
        // 各向异性过滤
        if (anisotropy > 1 && this.gfx.extensions.anisotropic) {
            gl.texParameterf(
                gl.TEXTURE_2D,
                this.gfx.extensions.anisotropic.TEXTURE_MAX_ANISOTROPY_EXT,
                Math.min(anisotropy, this.gfx.capabilities.anisotropy)
            );
        }
        
        // 上传纹理数据
        if (data) {
            gl.texImage2D(
                gl.TEXTURE_2D,
                0,
                glFormat.internalFormat,
                width,
                height,
                0,
                glFormat.format,
                glFormat.type,
                data
            );
        } else {
            gl.texImage2D(
                gl.TEXTURE_2D,
                0,
                glFormat.internalFormat,
                width,
                height,
                0,
                glFormat.format,
                glFormat.type,
                null
            );
        }
        
        // 生成mipmaps
        if (mipmaps && (width & (width - 1)) === 0 && (height & (height - 1)) === 0) {
            gl.generateMipmap(gl.TEXTURE_2D);
        }
        
        gl.bindTexture(gl.TEXTURE_2D, null);
        
        const id = this.gfx._createResource('texture', { 
            texture, 
            format: glFormat,
            width,
            height,
            wrap,
            filter
        });
        
        this.gfx._stats.textures++;
        return id;
    }
    
    mapTextureUsage(usage) {
        const usageMap = {
            'sampled': GPUTextureUsage.TEXTURE_BINDING | GPUTextureUsage.COPY_DST,
            'storage': GPUTextureUsage.STORAGE_BINDING | GPUTextureUsage.COPY_DST,
            'render': GPUTextureUsage.RENDER_ATTACHMENT,
            'copy': GPUTextureUsage.COPY_SRC | GPUTextureUsage.COPY_DST
        };
        
        if (this.rendererType === 'webgpu') {
            return usageMap[usage] || usageMap.sampled;
        }
        
        return usage;
    }
    
    mapTextureFormat(format) {
        const formatMap = {
            // WebGPU格式
            'rgba8unorm': { webgpu: 'rgba8unorm', webgl: null },
            'rgba8snorm': { webgpu: 'rgba8snorm', webgl: null },
            'rgba16float': { webgpu: 'rgba16float', webgl: null },
            'rgba32float': { webgpu: 'rgba32float', webgl: null },
            
            // WebGL格式
            'rgba': { 
                webgpu: 'rgba8unorm', 
                webgl: { 
                    internalFormat: this.context.RGBA, 
                    format: this.context.RGBA, 
                    type: this.context.UNSIGNED_BYTE 
                }
            },
            'rgb': { 
                webgpu: 'rgba8unorm',
                webgl: { 
                    internalFormat: this.context.RGB, 
                    format: this.context.RGB, 
                    type: this.context.UNSIGNED_BYTE 
                }
            },
            'depth': {
                webgpu: 'depth32float',
                webgl: {
                    internalFormat: this.context.DEPTH_COMPONENT,
                    format: this.context.DEPTH_COMPONENT,
                    type: this.context.UNSIGNED_INT
                }
            }
        };
        
        const mapped = formatMap[format] || formatMap.rgba;
        return this.rendererType === 'webgpu' ? mapped.webgpu : mapped.webgl;
    }
    
    mapTextureWrap(wrap) {
        const gl = this.context;
        const wrapMap = {
            'repeat': gl.REPEAT,
            'clamp': gl.CLAMP_TO_EDGE,
            'mirror': gl.MIRRORED_REPEAT
        };
        return wrapMap[wrap] || gl.REPEAT;
    }
    
    mapTextureFilter(filter) {
        const gl = this.context;
        const filterMap = {
            'nearest': gl.NEAREST,
            'linear': gl.LINEAR,
            'nearest-mipmap-nearest': gl.NEAREST_MIPMAP_NEAREST,
            'linear-mipmap-nearest': gl.LINEAR_MIPMAP_NEAREST,
            'nearest-mipmap-linear': gl.NEAREST_MIPMAP_LINEAR,
            'linear-mipmap-linear': gl.LINEAR_MIPMAP_LINEAR
        };
        return filterMap[filter] || gl.LINEAR;
    }
    
    uploadWebGPUTextureData(texture, data, width, height, format) {
        const { device } = this.context;
        
        const bytesPerRow = width * 4; // 假设RGBA 8位每通道
        
        device.queue.writeTexture(
            { texture },
            data,
            { bytesPerRow },
            { width, height }
        );
    }
    
    destroy(textureId) {
        this.gfx._destroyResource(textureId);
        this.gfx._stats.textures--;
    }
}
```

```javascript
// lib/shader.js
class ShaderManager {
    constructor(gfx) {
        this.gfx = gfx;
        this.context = gfx.context;
        this.rendererType = gfx.rendererType;
    }
    
    create(spec) {
        const { vert, frag, attributes, uniforms, topology = 'triangle-list' } = spec;
        
        if (this.rendererType === 'webgpu') {
            return this.createWebGPUShader(vert, frag, attributes, uniforms, topology);
        } else {
            return this.createWebGLShader(vert, frag, attributes, uniforms, topology);
        }
    }
    
    createWebGPUShader(vert, frag, attributes, uniforms, topology) {
        const { device } = this.context;
        
        // 编译着色器模块
        const vertexModule = device.createShaderModule({
            code: vert
        });
        
        const fragmentModule = device.createShaderModule({
            code: frag
        });
        
        // 创建管线布局
        const bindGroupLayouts = this.createWebGPUBindGroupLayouts(uniforms);
        const pipelineLayout = device.createPipelineLayout({
            bindGroupLayouts
        });
        
        // 创建渲染管线
        const pipeline = device.createRenderPipeline({
            layout: pipelineLayout,
            vertex: {
                module: vertexModule,
                entryPoint: 'main',
                buffers: this.createWebGPUVertexBuffers(attributes)
            },
            fragment: {
                module: fragmentModule,
                entryPoint: 'main',
                targets: [{
                    format: this.context.format
                }]
            },
            primitive: {
                topology: this.mapTopology(topology),
                cullMode: 'back'
            },
            depthStencil: {
                format: 'depth24plus',
                depthWriteEnabled: true,
                depthCompare: 'less'
            }
        });
        
        const id = this.gfx._createResource('pipeline', {
            pipeline,
            layout: pipelineLayout,
            bindGroupLayouts,
            vertexModule,
            fragmentModule,
            attributes,
            uniforms
        });
        
        this.gfx._stats.shaders++;
        return id;
    }
    
    createWebGLShader(vert, frag, attributes, uniforms, topology) {
        const gl = this.context;
        
        // 编译顶点着色器
        const vertexShader = gl.createShader(gl.VERTEX_SHADER);
        gl.shaderSource(vertexShader, vert);
        gl.compileShader(vertexShader);
        
        if (!gl.getShaderParameter(vertexShader, gl.COMPILE_STATUS)) {
            const error = gl.getShaderInfoLog(vertexShader);
            gl.deleteShader(vertexShader);
            throw new Error(`Vertex shader compilation failed: ${error}`);
        }
        
        // 编译片段着色器
        const fragmentShader = gl.createShader(gl.FRAGMENT_SHADER);
        gl.shaderSource(fragmentShader, frag);
        gl.compileShader(fragmentShader);
        
        if (!gl.getShaderParameter(fragmentShader, gl.COMPILE_STATUS)) {
            const error = gl.getShaderInfoLog(fragmentShader);
            gl.deleteShader(vertexShader);
            gl.deleteShader(fragmentShader);
            throw new Error(`Fragment shader compilation failed: ${error}`);
        }
        
        // 链接程序
        const program = gl.createProgram();
        gl.attachShader(program, vertexShader);
        gl.attachShader(program, fragmentShader);
        gl.linkProgram(program);
        
        if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
            const error = gl.getProgramInfoLog(program);
            gl.deleteShader(vertexShader);
            gl.deleteShader(fragmentShader);
            gl.deleteProgram(program);
            throw new Error(`Program linking failed: ${error}`);
        }
        
        // 清理着色器对象
        gl.deleteShader(vertexShader);
        gl.deleteShader(fragmentShader);
        
        // 获取attribute和uniform位置
        const attributeLocations = {};
        const uniformLocations = {};
        
        for (const name in attributes) {
            attributeLocations[name] = gl.getAttribLocation(program, name);
        }
        
        for (const name in uniforms) {
            uniformLocations[name] = gl.getUniformLocation(program, name);
        }
        
        const id = this.gfx._createResource('program', {
            program,
            attributeLocations,
            uniformLocations,
            attributes,
            uniforms,
            topology: this.mapTopology(topology)
        });
        
        this.gfx._stats.shaders++;
        return id;
    }
    
    createWebGPUBindGroupLayouts(uniforms) {
        const { device } = this.context;
        const entries = [];
        let bindingIndex = 0;
        
        for (const name in uniforms) {
            const uniform = uniforms[name];
            
            if (uniform.type === 'buffer') {
                entries.push({
                    binding: bindingIndex++,
                    visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
                    buffer: { type: 'uniform' }
                });
            } else if (uniform.type === 'sampler') {
                entries.push({
                    binding: bindingIndex++,
                    visibility: GPUShaderStage.FRAGMENT,
                    sampler: { type: 'filtering' }
                });
            } else if (uniform.type === 'texture') {
                entries.push({
                    binding: bindingIndex++,
                    visibility: GPUShaderStage.FRAGMENT,
                    texture: { sampleType: 'float' }
                });
            }
        }
        
        return [device.createBindGroupLayout({ entries })];
    }
    
    createWebGPUVertexBuffers(attributes) {
        const buffers = [];
        
        for (const name in attributes) {
            const attr = attributes[name];
            
            buffers.push({
                arrayStride: attr.stride || 0,
                stepMode: attr.stepMode || 'vertex',
                attributes: [{
                    shaderLocation: attr.location || 0,
                    offset: attr.offset || 0,
                    format: this.mapVertexFormat(attr.format)
                }]
            });
        }
        
        return buffers;
    }
    
    mapTopology(topology) {
        const topologyMap = {
            'point-list': { webgpu: 'point-list', webgl: 'POINTS' },
            'line-list': { webgpu: 'line-list', webgl: 'LINES' },
            'line-strip': { webgpu: 'line-strip', webgl: 'LINE_STRIP' },
            'triangle-list': { webgpu: 'triangle-list', webgl: 'TRIANGLES' },
            'triangle-strip': { webgpu: 'triangle-strip', webgl: 'TRIANGLE_STRIP' }
        };
        
        const mapped = topologyMap[topology] || topologyMap['triangle-list'];
        return this.rendererType === 'webgpu' ? mapped.webgpu : mapped.webgl;
    }
    
    mapVertexFormat(format) {
        const formatMap = {
            'float32': 'float32',
            'float32x2': 'float32x2',
            'float32x3': 'float32x3',
            'float32x4': 'float32x4',
            'uint32': 'uint32',
            'uint32x2': 'uint32x2',
            'uint32x3': 'uint32x3',
            'uint32x4': 'uint32x4'
        };
        
        return formatMap[format] || 'float32';
    }
    
    destroy(shaderId) {
        this.gfx._destroyResource(shaderId);
        this.gfx._stats.shaders--;
    }
}
```

```javascript
// lib/draw.js
class DrawManager {
    constructor(gfx) {
        this.gfx = gfx;
        this.context = gfx.context;
        this.rendererType = gfx.rendererType;
    }
    
    draw(spec) {
        const {
            pipeline,
            attributes,
            uniforms,
            indices,
            count,
            instances = 1,
            offset = 0,
            viewport = null,
            scissor = null
        } = spec;
        
        if (this.rendererType === 'webgpu') {
            this.drawWebGPU(pipeline, attributes, uniforms, indices, count, instances, offset, viewport, scissor);
        } else {
            this.drawWebGL(pipeline, attributes, uniforms, indices, count, instances, offset, viewport, scissor);
        }
        
        this.gfx._stats.drawCalls++;
        if (count) {
            this.gfx._stats.triangles += Math.floor(count / 3) * instances;
        }
    }
    
    drawWebGPU(pipelineId, attributes, uniforms, indices, count, instances, offset, viewport, scissor) {
        const { device, context } = this.context;
        const pipelineRes = this.gfx._getResource(pipelineId);
        
        if (!pipelineRes || pipelineRes.type !== 'pipeline') {
            throw new Error(`Pipeline ${pipelineId} not found`);
        }
        
        const { pipeline, layout } = pipelineRes.spec;
        const commandEncoder = device.createCommandEncoder();
        const textureView = context.getCurrentTexture().createView();
        
        // 创建渲染通道
        const passEncoder = commandEncoder.beginRenderPass({
            colorAttachments: [{
                view: textureView,
                loadOp: 'clear',
                storeOp: 'store'
            }]
        });
        
        // 设置视口和裁剪
        if (viewport) {
            passEncoder.setViewport(
                viewport.x, viewport.y, 
                viewport.width, viewport.height,
                viewport.minDepth || 0, viewport.maxDepth || 1
            );
        }
        
        if (scissor) {
            passEncoder.setScissorRect(
                scissor.x, scissor.y,
                scissor.width, scissor.height
            );
        }
        
        // 设置管线
        passEncoder.setPipeline(pipeline);
        
        // 绑定顶点缓冲区
        let vertexBufferIndex = 0;
        for (const name in attributes) {
            const attr = attributes[name];
            const bufferRes = this.gfx._getResource(attr.buffer);
            
            if (bufferRes && bufferRes.type === 'buffer') {
                passEncoder.setVertexBuffer(vertexBufferIndex++, bufferRes.spec.buffer);
            }
        }
        
        // 绑定索引缓冲区
        if (indices) {
            const indexRes = this.gfx._getResource(indices.buffer);
            if (indexRes && indexRes.type === 'buffer') {
                passEncoder.setIndexBuffer(
                    indexRes.spec.buffer,
                    indices.format === 'uint16' ? 'uint16' : 'uint32'
                );
            }
        }
        
        // 绑定uniforms
        if (uniforms && Object.keys(uniforms).length > 0) {
            const bindGroup = this.createWebGPUBindGroup(pipelineRes, uniforms);
            passEncoder.setBindGroup(0, bindGroup);
        }
        
        // 绘制
        if (indices) {
            passEncoder.drawIndexed(
                count || 0,
                instances,
                offset,
                0, // vertex offset
                0  // instance offset
            );
        } else {
            passEncoder.draw(
                count || 0,
                instances,
                offset,
                0  // instance offset
            );
        }
        
        passEncoder.end();
        device.queue.submit([commandEncoder.finish()]);
    }
    
    drawWebGL(pipelineId, attributes, uniforms, indices, count, instances, offset, viewport, scissor) {
        const gl = this.context;
        const programRes = this.gfx._getResource(pipelineId);
        
        if (!programRes || programRes.type !== 'program') {
            throw new Error(`Program ${pipelineId} not found`);
        }
        
        const { program, attributeLocations, uniformLocations } = programRes.spec;
        
        // 使用程序
        gl.useProgram(program);
        
        // 设置视口和裁剪
        if (viewport) {
            gl.viewport(viewport.x, viewport.y, viewport.width, viewport.height);
        }
        
        if (scissor) {
            gl.enable(gl.SCISSOR_TEST);
            gl.scissor(scissor.x, scissor.y, scissor.width, scissor.height);
        } else {
            gl.disable(gl.SCISSOR_TEST);
        }
        
        // 绑定顶点属性
        for (const name in attributes) {
            const attr = attributes[name];
            const location = attributeLocations[name];
            
            if (location === -1) continue;
            
            const bufferRes = this.gfx._getResource(attr.buffer);
            if (!bufferRes || bufferRes.type !== 'buffer') continue;
            
            const { buffer, target } = bufferRes.spec;
            
            gl.bindBuffer(target, buffer);
            gl.enableVertexAttribArray(location);
            gl.vertexAttribPointer(
                location,
                attr.size,
                attr.type || gl.FLOAT,
                attr.normalized || false,
                attr.stride || 0,
                attr.offset || 0
            );
            
            // 实例化渲染
            if (instances > 1 && this.gfx.extensions.instancedArrays) {
                this.gfx.extensions.instancedArrays.vertexAttribDivisorANGLE(
                    location,
                    attr.divisor || 0
                );
            }
        }
        
        // 绑定uniforms
        for (const name in uniforms) {
            const uniform = uniforms[name];
            const location = uniformLocations[name];
            
            if (!location) continue;
            
            if (uniform.type === 'matrix4fv') {
                gl.uniformMatrix4fv(location, false, uniform.value);
            } else if (uniform.type === 'matrix3fv') {
                gl.uniformMatrix3fv(location, false, uniform.value);
            } else if (uniform.type === 'vector3fv') {
                gl.uniform3fv(location, uniform.value);
            } else if (uniform.type === 'vector4fv') {
                gl.uniform4fv(location, uniform.value);
            } else if (uniform.type === 'float') {
                gl.uniform1f(location, uniform.value);
            } else if (uniform.type === 'int') {
                gl.uniform1i(location, uniform.value);
            } else if (uniform.type === 'sampler2D') {
                const textureRes = this.gfx._getResource(uniform.value);
                if (textureRes && textureRes.type === 'texture') {
                    gl.activeTexture(gl.TEXTURE0 + (uniform.unit || 0));
                    gl.bindTexture(gl.TEXTURE_2D, textureRes.spec.texture);
                    gl.uniform1i(location, uniform.unit || 0);
                }
            }
        }
        
        // 绑定索引缓冲区
        if (indices) {
            const indexRes = this.gfx._getResource(indices.buffer);
            if (indexRes && indexRes.type === 'buffer') {
                gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, indexRes.spec.buffer);
            }
        }
        
        // 绘制
        const mode = programRes.spec.topology;
        
        if (indices) {
            const type = indices.type || gl.UNSIGNED_SHORT;
            
            if (instances > 1 && this.gfx.extensions.instancedArrays) {
                this.gfx.extensions.instancedArrays.drawElementsInstancedANGLE(
                    mode,
                    count || 0,
                    type,
                    offset,
                    instances
                );
            } else {
                gl.drawElements(mode, count || 0, type, offset);
            }
        } else {
            if (instances > 1 && this.gfx.extensions.instancedArrays) {
                this.gfx.extensions.instancedArrays.drawArraysInstancedANGLE(
                    mode,
                    offset,
                    count || 0,
                    instances
                );
            } else {
                gl.drawArrays(mode, offset, count || 0);
            }
        }
        
        // 清理
        for (const name in attributes) {
            const location = attributeLocations[name];
            if (location !== -1) {
                gl.disableVertexAttribArray(location);
            }
        }
        
        gl.bindBuffer(gl.ARRAY_BUFFER, null);
        gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, null);
        gl.useProgram(null);
    }
    
    createWebGPUBindGroup(pipelineRes, uniforms) {
        const { device } = this.context;
        const { bindGroupLayouts } = pipelineRes.spec;
        
        const entries = [];
        let bindingIndex = 0;
        
        for (const name in uniforms) {
            const uniform = uniforms[name];
            
            if (uniform.type === 'buffer') {
                const bufferRes = this.gfx._getResource(uniform.value);
                if (bufferRes && bufferRes.type === 'buffer') {
                    entries.push({
                        binding: bindingIndex++,
                        resource: { buffer: bufferRes.spec.buffer }
                    });
                }
            } else if (uniform.type === 'sampler') {
                const samplerRes = this.gfx._getResource(uniform.value) || 
                    { spec: { sampler: this.gfx._defaultSampler } };
                entries.push({
                    binding: bindingIndex++,
                    resource: samplerRes.spec.sampler
                });
            } else if (uniform.type === 'texture') {
                const textureRes = this.gfx._getResource(uniform.value);
                if (textureRes && textureRes.type === 'texture') {
                    entries.push({
                        binding: bindingIndex++,
                        resource: textureRes.spec.texture.createView()
                    });
                }
            }
        }
        
        return device.createBindGroup({
            layout: bindGroupLayouts[0],
            entries
        });
    }
}
```

```html
                        vPosition = worldPos.xyz;
                        vNormal = mat3(model) * normal;
                        vTexcoord = texcoord;
                        
                        // 简单的动画
                        float offset = sin(time + position.x * 2.0) * 0.1;
                        worldPos.y += offset;
                        
                        gl_Position = projection * view * worldPos;
                    }
                `,
                frag: `
                    #version 300 es
                    precision highp float;
                    
                    in vec3 vNormal;
                    in vec2 vTexcoord;
                    in vec3 vPosition;
                    
                    uniform sampler2D texture;
                    uniform vec3 lightDir;
                    uniform vec3 cameraPos;
                    
                    out vec4 fragColor;
                    
                    void main() {
                        vec3 color = texture(texture, vTexcoord).rgb;
                        
                        // 简单光照
                        vec3 N = normalize(vNormal);
                        vec3 L = normalize(lightDir);
                        vec3 V = normalize(cameraPos - vPosition);
                        vec3 H = normalize(L + V);
                        
                        float diff = max(dot(N, L), 0.0);
                        float spec = pow(max(dot(N, H), 0.0), 32.0);
                        
                        vec3 ambient = color * 0.1;
                        vec3 diffuse = color * diff;
                        vec3 specular = vec3(0.3) * spec;
                        
                        fragColor = vec4(ambient + diffuse + specular, 1.0);
                    }
                `,
                attributes: {
                    position: {
                        buffer: cube.positions,
                        size: 3
                    },
                    normal: {
                        buffer: cube.normals,
                        size: 3
                    },
                    texcoord: {
                        buffer: cube.texcoords,
                        size: 2
                    }
                },
                uniforms: {
                    model: { type: 'matrix4fv', value: gfx.mat4.create() },
                    view: { type: 'matrix4fv', value: gfx.mat4.create() },
                    projection: { type: 'matrix4fv', value: gfx.mat4.create() },
                    time: { type: 'float', value: 0 },
                    texture: { type: 'sampler2D', value: null },
                    lightDir: { type: 'vector3fv', value: [0.5, 1.0, 0.5] },
                    cameraPos: { type: 'vector3fv', value: [0, 0, 5] }
                },
                indices: cube.indices,
                count: cube.count,
                primitive: wireframe ? 'line-strip' : 'triangle-list',
                depth: { enable: true },
                cull: { enable: true }
            });
            
            // 创建纹理
            const texture = gfx.createTexture2D(null, 256, 256, {
                wrap: 'repeat',
                filter: 'linear'
            });
            
            // 填充纹理数据
            const textureData = new Uint8Array(256 * 256 * 4);
            for (let i = 0; i < textureData.length; i += 4) {
                textureData[i] = Math.random() * 255;     // R
                textureData[i + 1] = Math.random() * 255; // G
                textureData[i + 2] = Math.random() * 255; // B
                textureData[i + 3] = 255;                 // A
            }
            gfx.textures.update(texture, textureData);
            
            // 开始渲染循环
            let time = 0;
            gfx.frame(() => {
                time += 0.01;
                
                gfx.clear([0.1, 0.1, 0.1, 1], 1);
                
                // 更新uniforms
                const model = gfx.mat4.identity(gfx.mat4.create());
                gfx.mat4.rotateY(model, model, time);
                
                const view = gfx.mat4.lookAt(
                    gfx.mat4.create(),
                    [Math.sin(time) * 5, 2, Math.cos(time) * 5],
                    [0, 0, 0],
                    [0, 1, 0]
                );
                
                const projection = gfx.mat4.perspective(
                    gfx.mat4.create(),
                    Math.PI / 4,
                    canvas.width / canvas.height,
                    0.1,
                    100
                );
                
                render({
                    uniforms: {
                        model,
                        view,
                        projection,
                        time,
                        texture,
                        cameraPos: [Math.sin(time) * 5, 2, Math.cos(time) * 5]
                    }
                });
                
                // 显示信息
                const stats = gfx.getStats();
                document.getElementById('info').textContent = 
                    `Renderer: ${gfx.rendererType.toUpperCase()} | ` +
                    `FPS: ${Math.round(1000 / (performance.now() - lastTime))} | ` +
                    `Draw Calls: ${stats.drawCalls} | ` +
                    `Triangles: ${stats.triangles}`;
                
                lastTime = performance.now();
            });
        }
        
        function toggleRenderer() {
            // 切换渲染器实现
            console.log('Renderer toggle not implemented');
        }
        
        function toggleWireframe() {
            wireframe = !wireframe;
            if (render) {
                render.spec.primitive = wireframe ? 'line-strip' : 'triangle-list';
            }
        }
        
        let lastTime = performance.now();
        init().catch(console.error);
    </script>
</body>
```

## 8. 特性总结

这个GFX库提供了：

### 🎯 **核心特性**
1. **双后端支持**：WebGL 2.0 + WebGPU，自动降级
2. **声明式API**：REGL风格的简洁API设计
3. **资源管理**：自动引用计数和垃圾回收
4. **类型安全**：TypeScript友好的设计

### 🛠 **功能模块**
1. **缓冲区管理**：顶点、索引、Uniform缓冲区
2. **纹理管理**：2D纹理、mipmap、各向异性过滤
3. **着色器编译**：GLSL 3.0 ES + WGSL支持
4. **渲染管线**：可配置的渲染状态
5. **几何体**：内置立方体、四边形、球体等

### ⚡ **性能优化**
1. **批处理**：自动批处理draw calls
2. **状态缓存**：减少GPU状态切换
3. **资源重用**：智能资源复用
4. **内存管理**：避免内存泄漏

### 🎨 **高级特性**
1. **实例化渲染**：支持硬件实例化
2. **计算着色器**：WebGPU计算管线
3. **多目标渲染**：MRT支持
4. **后期处理**：FrameBuffer链式处理
5. **调试工具**：性能统计和调试信息

### 📦 **模块化设计**
```
gfx/
├── core.js          # 核心上下文管理
├── buffer.js       # 缓冲区管理
├── texture.js      # 纹理管理
├── shader.js       # 着色器编译
├── draw.js         # 渲染命令
├── pipeline.js     # 渲染管线
├── geometry.js     # 几何体生成
├── math.js         # 数学库
└── gfx.js          # 高级API
```

这个库的设计目标是提供简单、高效、跨平台的图形编程体验，同时保持对底层API的良好封装和性能优化。
