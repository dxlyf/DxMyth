```typescript
interface DataPoint {
  x: number;
  y: number;
  dy?: number; // 一阶导数（可选）
  d2y?: number; // 二阶导数（可选）
}

interface HermitePolynomial {
  coefficients: number[];
  degree: number;
  evaluate: (x: number) => number;
  derivative: (order?: number) => (x: number) => number;
}

/**
 * 埃尔米特插值 (Hermite Interpolation)
 * 特点：不仅匹配函数值，还匹配导数值
 */
export class HermiteInterpolation {
  /**
   * 基础埃尔米特插值公式
   * 给定n+1个数据点，每个点有函数值和导数值
   */
  
  /**
   * 方法1：使用牛顿差商形式的埃尔米特插值
   */
  static interpolateNewton(data: DataPoint[]): HermitePolynomial {
    // 扩展数据点：对于有导数的点，重复该点
    const extendedNodes: { x: number; y: number; order: number }[] = [];
    
    for (const point of data) {
      extendedNodes.push({ x: point.x, y: point.y, order: 0 });
      
      if (point.dy !== undefined) {
        extendedNodes.push({ x: point.x, y: point.y, order: 1 });
      }
      
      if (point.d2y !== undefined) {
        extendedNodes.push({ x: point.x, y: point.y, order: 2 });
      }
    }
    
    // 按x值排序
    extendedNodes.sort((a, b) => a.x - b.x);
    
    const n = extendedNodes.length - 1;
    const xValues = extendedNodes.map(node => node.x);
    
    // 计算差商表
    const dividedDiff: number[][] = Array(n + 1)
      .fill(0)
      .map(() => Array(n + 1).fill(0));
    
    // 第一列是函数值
    for (let i = 0; i <= n; i++) {
      dividedDiff[i][0] = extendedNodes[i].y;
    }
    
    // 填充差商表（处理重复节点）
    for (let j = 1; j <= n; j++) {
      for (let i = 0; i <= n - j; i++) {
        const xi = xValues[i];
        const xij = xValues[i + j];
        
        if (Math.abs(xi - xij) < 1e-10) {
          // 重复节点，使用导数
          const node = extendedNodes.find(n => Math.abs(n.x - xi) < 1e-10)!;
          
          if (j === 1 && node.order >= 1) {
            dividedDiff[i][j] = node.dy!;
          } else if (j === 2 && node.order >= 2) {
            dividedDiff[i][j] = node.d2y! / 2;
          } else {
            dividedDiff[i][j] = 0;
          }
        } else {
          dividedDiff[i][j] = (dividedDiff[i + 1][j - 1] - dividedDiff[i][j - 1]) / (xij - xi);
        }
      }
    }
    
    // 构造牛顿形式的插值多项式
    const coefficients: number[] = [dividedDiff[0][0]];
    
    // 基函数系数
    for (let j = 1; j <= n; j++) {
      coefficients[j] = dividedDiff[0][j];
    }
    
    // 创建多项式求值函数
    const evaluate = (x: number): number => {
      let result = coefficients[0];
      let product = 1;
      
      for (let i = 1; i <= n; i++) {
        product *= (x - xValues[i - 1]);
        result += coefficients[i] * product;
      }
      
      return result;
    };
    
    // 导数函数
    const derivative = (order: number = 1): ((x: number) => number) => {
      return (x: number) => {
        // 数值导数
        const h = 1e-6;
        if (order === 1) {
          return (evaluate(x + h) - evaluate(x - h)) / (2 * h);
        } else {
          return (evaluate(x + h) - 2 * evaluate(x) + evaluate(x - h)) / (h * h);
        }
      };
    };
    
    return {
      coefficients,
      degree: n,
      evaluate,
      derivative
    };
  }
  
  /**
   * 方法2：使用拉格朗日形式的埃尔米特插值
   * 公式：H(x) = Σ [h_i(x)f(x_i) + k_i(x)f'(x_i)]
   */
  static interpolateLagrange(data: DataPoint[]): HermitePolynomial {
    const n = data.length;
    
    // 构造基函数
    const lagrangeBasis = (i: number): ((x: number) => number) => {
      return (x: number) => {
        let product = 1;
        const xi = data[i].x;
        
        for (let j = 0; j < n; j++) {
          if (i !== j) {
            const xj = data[j].x;
            product *= (x - xj) / (xi - xj);
          }
        }
        
        return product;
      };
    };
    
    // 构造h_i(x)和k_i(x)
    const hFunctions: ((x: number) => number)[] = [];
    const kFunctions: ((x: number) => number)[] = [];
    
    for (let i = 0; i < n; i++) {
      const Li = lagrangeBasis(i);
      const xi = data[i].x;
      
      // 计算L_i'(x_i)
      let LiPrimeXi = 0;
      for (let j = 0; j < n; j++) {
        if (i !== j) {
          let term = 1;
          for (let k = 0; k < n; k++) {
            if (k !== i && k !== j) {
              term *= (xi - data[k].x) / (xi - data[k].x);
            }
          }
          LiPrimeXi += term / (xi - data[j].x);
        }
      }
      
      // h_i(x) = [1 - 2(x - x_i)L_i'(x_i)] * [L_i(x)]^2
      const hi = (x: number) => {
        const Lix = Li(x);
        return (1 - 2 * (x - xi) * LiPrimeXi) * Lix * Lix;
      };
      
      // k_i(x) = (x - x_i) * [L_i(x)]^2
      const ki = (x: number) => (x - xi) * Li(x) * Li(x);
      
      hFunctions.push(hi);
      kFunctions.push(ki);
    }
    
    // 构造插值多项式
    const evaluate = (x: number): number => {
      let result = 0;
      
      for (let i = 0; i < n; i++) {
        const hi = hFunctions[i](x);
        const ki = kFunctions[i](x);
        result += hi * data[i].y;
        
        if (data[i].dy !== undefined) {
          result += ki * data[i].dy!;
        }
      }
      
      return result;
    };
    
    // 为了获取系数，可以在切比雪夫点上采样并拟合
    const degree = 2 * n - 1; // 埃尔米特插值多项式的最大次数
    const coefficients = this.fitPolynomial(evaluate, degree, -1, 1);
    
    const derivative = (order: number = 1): ((x: number) => number) => {
      return (x: number) => {
        const h = 1e-6;
        if (order === 1) {
          return (evaluate(x + h) - evaluate(x - h)) / (2 * h);
        } else {
          return (evaluate(x + h) - 2 * evaluate(x) + evaluate(x - h)) / (h * h);
        }
      };
    };
    
    return {
      coefficients,
      degree,
      evaluate,
      derivative
    };
  }
  
  /**
   * 方法3：分段三次埃尔米特插值
   * 最常用的埃尔米特插值形式
   */
  static piecewiseCubicHermite(data: DataPoint[]): {
    evaluate: (x: number) => number;
    derivative: (x: number) => number;
    segments: Array<{
      a: number; b: number; // 区间
      coefficients: number[]; // 三次多项式系数 [c0, c1, c2, c3]
    }>;
  } {
    if (data.length < 2) {
      throw new Error("至少需要2个数据点");
    }
    
    // 排序
    const sortedData = [...data].sort((a, b) => a.x - b.x);
    const segments: Array<{
      a: number; b: number;
      coefficients: number[];
    }> = [];
    
    // 为每个区间构造三次埃尔米特多项式
    for (let i = 0; i < sortedData.length - 1; i++) {
      const x0 = sortedData[i].x;
      const x1 = sortedData[i + 1].x;
      const y0 = sortedData[i].y;
      const y1 = sortedData[i + 1].y;
      
      // 导数（如果未提供，使用有限差分估计）
      let m0 = sortedData[i].dy;
      let m1 = sortedData[i + 1].dy;
      
      if (m0 === undefined || m1 === undefined) {
        // 使用中心差分估计导数
        if (i === 0) {
          m0 = (sortedData[i + 1].y - sortedData[i].y) / (x1 - x0);
        } else {
          m0 = (sortedData[i].y - sortedData[i - 1].y) / (x0 - sortedData[i - 1].x);
        }
        
        if (i === sortedData.length - 2) {
          m1 = (sortedData[i + 1].y - sortedData[i].y) / (x1 - x0);
        } else {
          m1 = (sortedData[i + 2].y - sortedData[i + 1].y) / (sortedData[i + 2].x - x1);
        }
      }
      
      // 三次埃尔米特插值公式
      // 在区间[0,1]上的参数t = (x - x0)/(x1 - x0)
      // 多项式：h00(t)*y0 + h10(t)*m0*h + h01(t)*y1 + h11(t)*m1*h
      // 其中 h = x1 - x0
      
      const h = x1 - x0;
      
      // 埃尔米特基函数：
      // h00(t) = 2t^3 - 3t^2 + 1
      // h10(t) = t^3 - 2t^2 + t
      // h01(t) = -2t^3 + 3t^2
      // h11(t) = t^3 - t^2
      
      // 转换为标准多项式形式：c0 + c1*t + c2*t^2 + c3*t^3
      // 其中 t = (x - x0)/h
      
      const c0 = y0;
      const c1 = m0 * h;
      const c2 = -3 * y0 + 3 * y1 - 2 * m0 * h - m1 * h;
      const c3 = 2 * y0 - 2 * y1 + m0 * h + m1 * h;
      
      segments.push({
        a: x0,
        b: x1,
        coefficients: [c0, c1, c2, c3]
      });
    }
    
    // 求值函数
    const evaluate = (x: number): number => {
      // 找到包含x的区间
      for (const segment of segments) {
        if (x >= segment.a && x <= segment.b) {
          const t = (x - segment.a) / (segment.b - segment.a);
          const [c0, c1, c2, c3] = segment.coefficients;
          
          // 使用霍纳法则求值
          return c0 + t * (c1 + t * (c2 + t * c3));
        }
      }
      
      // 外推：使用第一个或最后一个区间
      if (x < segments[0].a) {
        const segment = segments[0];
        const t = (x - segment.a) / (segment.b - segment.a);
        const [c0, c1, c2, c3] = segment.coefficients;
        return c0 + t * (c1 + t * (c2 + t * c3));
      } else {
        const segment = segments[segments.length - 1];
        const t = (x - segment.a) / (segment.b - segment.a);
        const [c0, c1, c2, c3] = segment.coefficients;
        return c0 + t * (c1 + t * (c2 + t * c3));
      }
    };
    
    // 导数函数
    const derivative = (x: number): number => {
      for (const segment of segments) {
        if (x >= segment.a && x <= segment.b) {
          const h = segment.b - segment.a;
          const t = (x - segment.a) / h;
          const [c0, c1, c2, c3] = segment.coefficients;
          
          // 对t求导，然后乘以dt/dx = 1/h
          const derivativeT = c1 + 2 * c2 * t + 3 * c3 * t * t;
          return derivativeT / h;
        }
      }
      
      return 0;
    };
    
    return {
      evaluate,
      derivative,
      segments
    };
  }
  
  /**
   * 方法4：带张力参数的三次埃尔米特插值
   * 常用于Catmull-Rom样条
   */
  static cubicHermiteWithTension(
    p0: Point2D, p1: Point2D, p2: Point2D, p3: Point2D,
    t: number,
    tension: number = 0.5
  ): Point2D {
    // 计算切线
    const m1 = {
      x: tension * (p2.x - p0.x),
      y: tension * (p2.y - p0.y)
    };
    
    const m2 = {
      x: tension * (p3.x - p1.x),
      y: tension * (p3.y - p1.y)
    };
    
    // 埃尔米特基函数
    const t2 = t * t;
    const t3 = t2 * t;
    
    const h1 = 2 * t3 - 3 * t2 + 1;  // h00
    const h2 = -2 * t3 + 3 * t2;     // h01
    const h3 = t3 - 2 * t2 + t;      // h10
    const h4 = t3 - t2;              // h11
    
    return {
      x: h1 * p1.x + h2 * p2.x + h3 * m1.x + h4 * m2.x,
      y: h1 * p1.y + h2 * p2.y + h3 * m1.y + h4 * m2.y
    };
  }
  
  /**
   * 辅助函数：通过采样拟合多项式系数
   */
  private static fitPolynomial(
    f: (x: number) => number,
    degree: number,
    a: number,
    b: number
  ): number[] {
    // 使用切比雪夫节点采样
    const n = degree + 1;
    const nodes: number[] = [];
    const values: number[] = [];
    
    for (let k = 0; k < n; k++) {
      const theta = Math.PI * (2 * k + 1) / (2 * n);
      const x = (a + b) / 2 + (b - a) / 2 * Math.cos(theta);
      nodes.push(x);
      values.push(f(x));
    }
    
    // 使用最小二乘法拟合
    const X: number[][] = [];
    for (const x of nodes) {
      const row: number[] = [];
      for (let i = 0; i <= degree; i++) {
        row.push(Math.pow(x, i));
      }
      X.push(row);
    }
    
    // 简化的最小二乘解
    const coefficients = this.solveLeastSquares(X, values, degree + 1);
    return coefficients;
  }
  
  private static solveLeastSquares(A: number[][], b: number[], n: number): number[] {
    // 构造正规方程 A^T A x = A^T b
    const AT: number[][] = [];
    for (let j = 0; j < n; j++) {
      AT[j] = [];
      for (let i = 0; i < A.length; i++) {
        AT[j][i] = A[i][j];
      }
    }
    
    // 计算 A^T A
    const ATA: number[][] = [];
    for (let i = 0; i < n; i++) {
      ATA[i] = [];
      for (let j = 0; j < n; j++) {
        let sum = 0;
        for (let k = 0; k < A.length; k++) {
          sum += AT[i][k] * A[k][j];
        }
        ATA[i][j] = sum;
      }
    }
    
    // 计算 A^T b
    const ATb: number[] = [];
    for (let i = 0; i < n; i++) {
      let sum = 0;
      for (let k = 0; k < A.length; k++) {
        sum += AT[i][k] * b[k];
      }
      ATb[i] = sum;
    }
    
    // 解线性方程组（使用高斯消元法简化）
    return this.solveLinearSystem(ATA, ATb);
  }
  
  private static solveLinearSystem(A: number[][], b: number[]): number[] {
    const n = A.length;
    const augmented = A.map((row, i) => [...row, b[i]]);
    
    // 高斯消元
    for (let i = 0; i < n; i++) {
      // 寻找主元
      let maxRow = i;
      for (let j = i + 1; j < n; j++) {
        if (Math.abs(augmented[j][i]) > Math.abs(augmented[maxRow][i])) {
          maxRow = j;
        }
      }
      
      // 交换行
      [augmented[i], augmented[maxRow]] = [augmented[maxRow], augmented[i]];
      
      // 归一化
      const pivot = augmented[i][i];
      if (Math.abs(pivot) < 1e-12) {
        throw new Error("矩阵奇异");
      }
      
      for (let j = i; j <= n; j++) {
        augmented[i][j] /= pivot;
      }
      
      // 消元
      for (let j = 0; j < n; j++) {
        if (j !== i) {
          const factor = augmented[j][i];
          for (let k = i; k <= n; k++) {
            augmented[j][k] -= factor * augmented[i][k];
          }
        }
      }
    }
    
    // 提取解
    return augmented.map(row => row[n]);
  }
}

/**

export class HermiteTheory {
  /**
   * 埃尔米特插值的基本思想
   */
  static basicConcept() {
    return {
      definition: "埃尔米特插值是一种插值方法，它不仅要求插值多项式在节点处与函数值相等，还要求导数（或高阶导数）相等。",
      
      problemStatement: `
        给定 n+1 个节点 x₀, x₁, ..., xₙ 和相应的函数值 f(xᵢ) 以及导数值 f'(xᵢ), f''(xᵢ), ...
        找到一个多项式 H(x)，使得：
        1. H(xᵢ) = f(xᵢ)
        2. H'(xᵢ) = f'(xᵢ)
        3. H''(xᵢ) = f''(xᵢ) （如果给定）
        ...
      `,
      
      uniqueness: "如果总条件数（函数值 + 导数值）为 m+1，则存在唯一的次数不超过 m 的多项式满足所有条件。",
      
      degree: "对于 n+1 个节点，如果每个节点有 kᵢ 个条件（包括函数值和导数值），则多项式次数不超过 Σkᵢ - 1。"
    };
  }
  
  /**
   * 主要公式
   */
  static formulas() {
    return {
      newtonForm: `
        牛顿差商形式：
        H(x) = f[x₀] + f[x₀,x₁](x-x₀) + f[x₀,x₁,x₂](x-x₀)² + ...
        对于重复节点，差商定义为：
        f[xᵢ,xᵢ] = f'(xᵢ)
        f[xᵢ,xᵢ,xᵢ] = f''(xᵢ)/2
      `,
      
      lagrangeForm: `
        拉格朗日形式（两点情况）：
        H(x) = h₀₀(x)f(x₀) + h₀₁(x)f(x₁) + h₁₀(x)f'(x₀) + h₁₁(x)f'(x₁)
        其中基函数为：
        h₀₀(x) = [1 + 2(x-x₀)/(x₁-x₀)] * [(x-x₁)/(x₀-x₁)]²
        h₀₁(x) = [1 + 2(x-x₁)/(x₀-x₁)] * [(x-x₀)/(x₁-x₀)]²
        h₁₀(x) = (x-x₀) * [(x-x₁)/(x₀-x₁)]²
        h₁₁(x) = (x-x₁) * [(x-x₀)/(x₁-x₀)]²
      `,
      
      cubicHermite: `
        分段三次埃尔米特插值（最常用）：
        在区间 [x₀, x₁] 上，令 t = (x-x₀)/(x₁-x₀)，则：
        H(x) = h₀₀(t)f(x₀) + h₀₁(t)f(x₁) + h₁₀(t)f'(x₀)(x₁-x₀) + h₁₁(t)f'(x₁)(x₁-x₀)
        其中：
        h₀₀(t) = 2t³ - 3t² + 1
        h₀₁(t) = -2t³ + 3t²
        h₁₀(t) = t³ - 2t² + t
        h₁₁(t) = t³ - t²
      `,
      
      errorFormula: `
        误差公式：
        f(x) - H(x) = f^{(m+1)}(ξ)/(m+1)! * Π(x-xᵢ)^{kᵢ}
        其中 m = Σkᵢ - 1，ξ 在节点最小值和最大值之间。
      `
    };
  }
```

## 埃尔米特插值公式详解

### 1. **基本概念**

埃尔米特插值是一种**带导数的插值方法**。它不仅要求插值多项式在节点处与函数值相等，还要求导数（或高阶导数）相等。

给定节点 $x_0, x_1, \ldots, x_n$，以及对应的：
- 函数值 $f(x_i)$
- 一阶导数值 $f'(x_i)$
- 二阶导数值 $f''(x_i)$（可选）
- ...

### 2. **主要公式形式**

#### **两点三次埃尔米特插值（最常用）**

对于区间 $[x_0, x_1]$，给定：
- $f(x_0) = y_0$, $f'(x_0) = m_0$
- $f(x_1) = y_1$, $f'(x_1) = m_1$

插值多项式为：
$$
H(x) = h_{00}(t)y_0 + h_{01}(t)y_1 + h_{10}(t)m_0h + h_{11}(t)m_1h
$$

其中 $t = \frac{x-x_0}{h}$, $h = x_1 - x_0$，基函数为：

$$
\begin{aligned}
h_{00}(t) &= 2t^3 - 3t^2 + 1 \\
h_{01}(t) &= -2t^3 + 3t^2 \\
h_{10}(t) &= t^3 - 2t^2 + t \\
h_{11}(t) &= t^3 - t^2
\end{aligned}
$$

#### **矩阵形式**

$$
\begin{bmatrix}
y_0 \\ y_1 \\ m_0 \\ m_1
\end{bmatrix}
=
\begin{bmatrix}
1 & 0 & 0 & 0 \\
0 & 1 & 0 & 0 \\
-\frac{3}{h^2} & \frac{3}{h^2} & -\frac{2}{h} & -\frac{1}{h} \\
\frac{2}{h^3} & -\frac{2}{h^3} & \frac{1}{h^2} & \frac{1}{h^2}
\end{bmatrix}
\begin{bmatrix}
a_0 \\ a_1 \\ a_2 \\ a_3
\end{bmatrix}
$$

其中 $H(x) = a_0 + a_1x + a_2x^2 + a_3x^3$

### 3. **牛顿差商形式**

对于有导数的节点，通过**重复节点**来处理：

设节点 $x_i$ 有 $k_i$ 个条件，则在差商表中重复 $x_i$ 共 $k_i$ 次：

$$
\begin{aligned}
f[x_i, x_i] &= f'(x_i) \\
f[x_i, x_i, x_i] &= \frac{f''(x_i)}{2} \\
f[x_i, x_i, x_i, x_i] &= \frac{f'''(x_i)}{6} \\
&\vdots
\end{aligned}
$$

$$
H(x)
