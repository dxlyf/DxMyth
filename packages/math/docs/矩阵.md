# Matrix3 列主序矩阵类实现

下面是一个完整的 Matrix3 矩阵类实现，采用**列主序(column-major)**存储方式，符合 WebGL/OpenGL 标准。

```javascript
class Vector2 {
  constructor(x = 0, y = 0) {
    this.x = x;
    this.y = y;
  }
  
  clone() {
    return new Vector2(this.x, this.y);
  }
  
  equals(v) {
    return this.x === v.x && this.y === v.y;
  }
}

class Matrix3 {
  constructor() {
    // 列主序存储
    this.elements = [
      1, 0, 0,  // 第一列
      0, 1, 0,  // 第二列
      0, 0, 1   // 第三列
    ];
  }

  // 设置矩阵元素 (参数按数学惯例 m11,m12,m13,m21...)
  set(m11, m12, m13, 
      m21, m22, m23, 
      m31, m32, m33) {
    const te = this.elements;
    // 列主序存储
    te[0] = m11; te[3] = m12; te[6] = m13;
    te[1] = m21; te[4] = m22; te[7] = m23;
    te[2] = m31; te[5] = m32; te[8] = m33;
    return this;
  }

  // 重置为单位矩阵
  identity() {
    this.set(
      1, 0, 0,
      0, 1, 0,
      0, 0, 1
    );
    return this;
  }

  // 复制矩阵
  copy(m) {
    const te = this.elements;
    const me = m.elements;
    for (let i = 0; i < 9; i++) {
      te[i] = me[i];
    }
    return this;
  }

  // 矩阵乘法 (列主序)
  multiply(m) {
    return this.multiplyMatrices(this, m);
  }

  // 预乘矩阵
  premultiply(m) {
    return this.multiplyMatrices(m, this);
  }

  // 列主序矩阵乘法
  multiplyMatrices(a, b) {
    const ae = a.elements;
    const be = b.elements;
    const te = this.elements;

    // 第一列
    const a11 = ae[0], a21 = ae[1], a31 = ae[2];
    const b11 = be[0], b21 = be[1], b31 = be[2];
    te[0] = a11 * b11 + a21 * be[3] + a31 * be[6];
    te[1] = a11 * b21 + a21 * be[4] + a31 * be[7];
    te[2] = a11 * b31 + a21 * be[5] + a31 * be[8];

    // 第二列
    const a12 = ae[3], a22 = ae[4], a32 = ae[5];
    te[3] = a12 * b11 + a22 * be[3] + a32 * be[6];
    te[4] = a12 * b21 + a22 * be[4] + a32 * be[7];
    te[5] = a12 * b31 + a22 * be[5] + a32 * be[8];

    // 第三列
    const a13 = ae[6], a23 = ae[7], a33 = ae[8];
    te[6] = a13 * b11 + a23 * be[3] + a33 * be[6];
    te[7] = a13 * b21 + a23 * be[4] + a33 * be[7];
    te[8] = a13 * b31 + a23 * be[5] + a33 * be[8];

    return this;
  }

  // 矩阵缩放 (列主序)
  scale(sx, sy) {
    const te = this.elements;
    // 第一列乘以sx
    te[0] *= sx; te[1] *= sx; te[2] *= sx;
    // 第二列乘以sy
    te[3] *= sy; te[4] *= sy; te[5] *= sy;
    return this;
  }

  // 矩阵旋转 (列主序)
  rotate(theta) {
    const c = Math.cos(theta);
    const s = Math.sin(theta);
    const te = this.elements;
    
    // 第一列
    const a11 = te[0], a21 = te[1];
    te[0] = c * a11 - s * a21;
    te[1] = s * a11 + c * a21;
    
    // 第二列
    const a12 = te[3], a22 = te[4];
    te[3] = c * a12 - s * a22;
    te[4] = s * a12 + c * a22;
    
    return this;
  }

  // 矩阵平移 (列主序)
  translate(tx, ty) {
    const te = this.elements;
    // 第三列 = 第三列 + 第一列*tx + 第二列*ty
    te[6] += te[0] * tx + te[3] * ty;
    te[7] += te[1] * tx + te[4] * ty;
    te[8] += te[2] * tx + te[5] * ty;
    return this;
  }

  // 矩阵斜切 (列主序)
  skew(skx, sky) {
    const te = this.elements;
    const a11 = te[0], a21 = te[1];
    const a12 = te[3], a22 = te[4];
    
    // 第一列 += 第二列 * skx
    te[0] += a12 * skx;
    te[1] += a22 * skx;
    
    // 第二列 += 第一列 * sky
    te[3] += a11 * sky;
    te[4] += a21 * sky;
    
    return this;
  }

  // 求逆矩阵 (列主序)
  invert() {
    const te = this.elements;
    // 第一列
    const n11 = te[0], n21 = te[1], n31 = te[2];
    // 第二列
    const n12 = te[3], n22 = te[4], n32 = te[5];
    // 第三列
    const n13 = te[6], n23 = te[7], n33 = te[8];
    
    // 计算行列式
    const t11 = n33 * n22 - n32 * n23;
    const t12 = n32 * n13 - n33 * n12;
    const t13 = n23 * n12 - n22 * n13;
    
    const det = n11 * t11 + n21 * t12 + n31 * t13;
    
    if (det === 0) return this.identity();
    
    const detInv = 1 / det;
    
    // 第一列
    te[0] = t11 * detInv;
    te[1] = (n31 * n23 - n33 * n21) * detInv;
    te[2] = (n32 * n21 - n31 * n22) * detInv;
    
    // 第二列
    te[3] = t12 * detInv;
    te[4] = (n33 * n11 - n31 * n13) * detInv;
    te[5] = (n31 * n12 - n32 * n11) * detInv;
    
    // 第三列
    te[6] = t13 * detInv;
    te[7] = (n21 * n13 - n23 * n11) * detInv;
    te[8] = (n22 * n11 - n21 * n12) * detInv;
    
    return this;
  }

  // 转置矩阵 (列主序转行主序)
  transpose() {
    const te = this.elements;
    let tmp;
    
    tmp = te[1]; te[1] = te[3]; te[3] = tmp;
    tmp = te[2]; te[2] = te[6]; te[6] = tmp;
    tmp = te[5]; te[5] = te[7]; te[7] = tmp;
    
    return this;
  }

  // 克隆矩阵
  clone() {
    return new Matrix3().copy(this);
  }

  // 从数组设置矩阵 (列主序)
  fromArray(array, offset = 0) {
    for (let i = 0; i < 9; i++) {
      this.elements[i] = array[i + offset];
    }
    return this;
  }

  // 转换为数组 (列主序)
  toArray(array = [], offset = 0) {
    const te = this.elements;
    for (let i = 0; i < 9; i++) {
      array[offset + i] = te[i];
    }
    return array;
  }

  // 判断矩阵是否相等
  equals(matrix) {
    const te = this.elements;
    const me = matrix.elements;
    for (let i = 0; i < 9; i++) {
      if (te[i] !== me[i]) return false;
    }
    return true;
  }

  // 创建一个包含平移、旋转、斜切、缩放、原点、枢轴的矩阵 (列主序)
  static makeTranslateRotationSkewScaleOriginPivot(translate, rotation, skew, scale, origin, pivot) {
    const matrix = new Matrix3();
    
    // 应用枢轴点 (先平移)
    if (pivot && (pivot.x !== 0 || pivot.y !== 0)) {
      matrix.translate(-pivot.x, -pivot.y);
    }
    
    // 应用缩放
    if (scale && (scale.x !== 1 || scale.y !== 1)) {
      matrix.scale(scale.x, scale.y);
    }
    
    // 应用斜切
    if (skew && (skew.x !== 0 || skew.y !== 0)) {
      matrix.skew(Math.tan(skew.x), Math.tan(skew.y));
    }
    
    // 应用旋转
    if (rotation !== 0) {
      matrix.rotate(rotation);
    }
    
    // 应用平移
    if (translate && (translate.x !== 0 || translate.y !== 0)) {
      matrix.translate(translate.x, translate.y);
    }
    
    // 应用原点
    if (origin && (origin.x !== 0 || origin.y !== 0)) {
      matrix.translate(origin.x, origin.y);
    }
    
    return matrix;
  }

  // 分解矩阵为平移、旋转、斜切、缩放、原点、枢轴 (列主序)
  decompose() {
    const te = this.elements;
    const result = {
      translate: new Vector2(),
      rotation: 0,
      skew: new Vector2(),
      scale: new Vector2(1, 1),
      origin: new Vector2(),
      pivot: new Vector2()
    };
    
    // 提取平移部分 (第三列的前两行)
    result.translate.x = te[6];
    result.translate.y = te[7];
    
    // 提取缩放和旋转部分 (第一、二列)
    const a = te[0], b = te[1];  // 第一列
    const c = te[3], d = te[4];  // 第二列
    
    // 计算缩放
    result.scale.x = Math.sqrt(a * a + b * b);
    result.scale.y = Math.sqrt(c * c + d * d);
    
    // 计算旋转角度 (从第一列)
    result.rotation = Math.atan2(b, a);
    
    // 计算斜切
    const sinRotation = Math.sin(result.rotation);
    const cosRotation = Math.cos(result.rotation);
    
    // 移除旋转影响后的矩阵
    const invScaleX = 1 / result.scale.x;
    const invScaleY = 1 / result.scale.y;
    
    const aPrime = a * invScaleX;
    const bPrime = b * invScaleX;
    const cPrime = c * invScaleY;
    const dPrime = d * invScaleY;
    
    // 计算斜切角度
    result.skew.x = Math.atan2(aPrime * sinRotation + cPrime * cosRotation,
                              aPrime * cosRotation - cPrime * sinRotation);
    result.skew.y = Math.atan2(bPrime * cosRotation + dPrime * sinRotation,
                              dPrime * cosRotation - bPrime * sinRotation);
    
    // 原点和枢轴需要更复杂的计算，这里简化处理
    result.origin.copy(result.translate);
    
    return result;
  }
}
```

## 关键改进说明

1. **列主序存储**：
   - 矩阵元素按列存储：`[col1, col2, col3]` = `[m11,m21,m31, m12,m22,m32, m13,m23,m33]`
   - 所有方法都适配了列主序计算逻辑

2. **数学一致性**：
   - 保持数学表示法 `m11`(第1行第1列)，`m12`(第1行第2列)等
   - 内部存储索引根据列主序调整

3. **WebGL兼容**：
   - 可直接用于 WebGL 的 `uniformMatrix3fv` 等函数
   - 变换顺序与图形API标准一致

4. **性能优化**：
   - 减少临时变量创建
   - 使用更高效的列主序矩阵乘法

这个实现完全采用列主序，适合需要与 WebGL/OpenGL 交互的场景，同时保持了清晰的数学接口。

# 计算机图形学核心数学：矩阵、四元数与投影

## 一、变换矩阵（2D & 3D）

### 1. 2D 变换矩阵（3×3）
```javascript
class Matrix2D {
  constructor() {
    this.m = [1, 0, 0,  // 第一列
              0, 1, 0,  // 第二列
              0, 0, 1]; // 第三列（齐次坐标）
  }

  // 平移矩阵
  static translate(tx, ty) {
    return new Matrix2D().set([
      1, 0, tx,
      0, 1, ty,
      0, 0, 1
    ]);
  }

  // 旋转矩阵（弧度）
  static rotate(θ) {
    const c = Math.cos(θ), s = Math.sin(θ);
    return new Matrix2D().set([
      c, -s, 0,
      s,  c, 0,
      0,  0, 1
    ]);
  }
}
```

### 2. 3D 变换矩阵（4×4）
```javascript
class Matrix4 {
  // 绕任意轴旋转（单位向量 axis = [x,y,z]）
  static rotateAxis(axis, θ) {
    const [x, y, z] = axis;
    const c = Math.cos(θ), s = Math.sin(θ);
    const t = 1 - c;
    
    return new Matrix4().set([
      t*x*x + c,   t*x*y - z*s, t*x*z + y*s, 0,
      t*y*x + z*s, t*y*y + c,   t*y*z - x*s, 0,
      t*z*x - y*s, t*z*y + x*s, t*z*z + c,   0,
      0, 0, 0, 1
    ]);
  }
}
```

```javascript
function inverse2D(m) {
  const det = m[0]*m[4] - m[1]*m[3];
```

```javascript
function ortho(left, right, bottom, top, near, far) {
```

```javascript
function perspective(fov, aspect, near, far) {
  const f = 1 / Math.tan(fov/2);
  return [
    f/aspect, 0, 0, 0,
    0, f, 0, 0,
    0, 0, (far+near)/(near-far), (2*far*near)/(near-far),
    0, 0, -1, 0
  ];
}
```

## 四、四元数（Quaternion）

### 1. 四元数旋转
```javascript
class Quaternion {
  constructor(x=0, y=0, z=0, w=1) {
    this.x = x; this.y = y; this.z = z; this.w = w;
  }

  // 从轴角转换
  static fromAxisAngle(axis, angle) {
    const halfAngle = angle/2;
    const s = Math.sin(halfAngle);
    return new Quaternion(
      axis[0] * s,
      axis[1] * s,
      axis[2] * s,
      Math.cos(halfAngle)
    );
  }

  // 转换为旋转矩阵
  toMatrix4() {
    const {x, y, z, w} = this;
    return [
      1-2*y*y-2*z*z, 2*x*y-2*w*z,   2*x*z+2*w*y,   0,
      2*x*y+2*w*z,   1-2*x*x-2*z*z, 2*y*z-2*w*x,   0,
      2*x*z-2*w*y,   2*y*z+2*w*x,   1-2*x*x-2*y*y, 0,
      0, 0, 0, 1
    ];
  }
}
```

## 五、欧拉角 ↔ 矩阵

### 1. 欧拉角转矩阵（YZX顺序）
```javascript
function eulerToMatrix(yaw, pitch, roll) {
  const cy = Math.cos(yaw), sy = Math.sin(yaw);
  const cp = Math.cos(pitch), sp = Math.sin(pitch);
  const cr = Math.cos(roll), sr = Math.sin(roll);

  return [
    cy*cr, cy*sr*sp - sy*cp, cy*sr*cp + sy*sp, 0,
    sy*cr, sy*sr*sp + cy*cp, sy*sr*cp - cy*sp, 0,
    -sr,   cr*sp,           cr*cp,           0,
    0, 0, 0, 1
  ];
}
```

### 2. 矩阵转欧拉角（提取YZX顺序）

```javascript
function matrixToEuler(m) {
  const pitch = Math.asin(-m[8]);
  const yaw = Math.atan2(m[0], m[4]);
  const roll = Math.atan2(m[9], m[10]);
  return { yaw, pitch, roll };
}
```

```javascript
function slerp(q1, q2, t) {
  const cosθ = q1.x*q2.x + q1.y*q2.y + q1.z*q2.z + q1.w*q2.w;
  const θ = Math.acos(cosθ);
  const sinθ = Math.sin(θ);
  
  const w1 = Math.sin((1-t)*θ) / sinθ;
  const w2 = Math.sin(t*θ) / sinθ;
  
  return new Quaternion(
    w1*q1.x + w2*q2.x,
    w1*q1.y + w2*q2.y,
    w1*q1.z + w2*q2.z,
    w1*q1.w + w2*q2.w
  );
}
```
