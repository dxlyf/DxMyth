# 曲线类型大全及数学表示

曲线在计算机图形学、CAD/CAM、数据拟合等领域有广泛应用。除了贝塞尔曲线，还有很多其他重要类型的曲线。

## 📊 主要曲线类型分类

| 曲线类型 | 主要特点 | 应用场景 |
|---------|---------|---------|
| **贝塞尔曲线** | 控制点定义，易于实现 | UI设计、字体渲染、简单路径 |
| **B样条曲线** | 局部控制，更平滑 | CAD/CAM、动画路径、复杂建模 |
| **NURBS曲线** | 非均匀有理B样条，最通用 | 工业设计、汽车造型、高级3D建模 |
| **埃尔米特曲线** | 端点位置和切线定义 | 物理模拟、运动规划 |
| **Catmull-Rom曲线** | 通过所有控制点 | 相机路径、平滑插值 |
| **多项式曲线** | 数学表达式定义 | 数据拟合、函数绘图 |

## 🧮 各种曲线的TypeScript实现

```typescript
// 向量和点类型定义
type Vector2 = [number, number];
type Vector3 = [number, number, number];
type Point = Vector2 | Vector3;

// 基础数学函数
class MathUtils {
  // 线性插值
  static lerp(a: number, b: number, t: number): number {
    return a + (b - a) * t;
  }

  // 阶乘
  static factorial(n: number): number {
    if (n <= 1) return 1;
    return n * this.factorial(n - 1);
  }

  // 二项式系数
  static binomial(n: number, k: number): number {
    if (k < 0 || k > n) return 0;
    return this.factorial(n) / (this.factorial(k) * this.factorial(n - k));
  }

  // 两点距离
  static distance(p1: Point, p2: Point): number {
    const dx = p2[0] - p1[0];
    const dy = p2[1] - p1[1];
    return Math.sqrt(dx * dx + dy * dy);
  }

  // 向量标准化
  static normalize(v: Point): Point {
    const length = this.distance([0, 0], v);
    if (length === 0) return v;
    return [v[0] / length, v[1] / length] as Point;
  }
}

// 曲线基类
abstract class Curve {
  abstract getPoint(t: number): Point;
  abstract getPoints(segments: number): Point[];
}

// ========== 贝塞尔曲线 ==========
class BezierCurve extends Curve {
  constructor(private controlPoints: Point[]) {
    super();
  }

  getPoint(t: number): Point {
    const n = this.controlPoints.length - 1;
    let result: Point = [0, 0];
    
    for (let i = 0; i <= n; i++) {
      const binomial = MathUtils.binomial(n, i);
      const term = binomial * Math.pow(1 - t, n - i) * Math.pow(t, i);
      
      result[0] += this.controlPoints[i][0] * term;
      result[1] += this.controlPoints[i][1] * term;
    }
    
    return result;
  }

  getPoints(segments: number = 100): Point[] {
    const points: Point[] = [];
    for (let i = 0; i <= segments; i++) {
      const t = i / segments;
      points.push(this.getPoint(t));
    }
    return points;
  }
}

// ========== B样条曲线 ==========
class BSplineCurve extends Curve {
  constructor(
    private controlPoints: Point[],
    private degree: number = 3,
    private knots?: number[]
  ) {
    super();
    if (!knots) {
      this.knots = this.generateKnots();
    }
  }

  private generateKnots(): number[] {
    const n = this.controlPoints.length;
    const knots: number[] = [];
    
    // 均匀节点向量
    for (let i = 0; i < n + this.degree + 1; i++) {
      knots.push(i);
    }
    
    return knots;
  }

  // B样条基函数
  private basisFunction(i: number, degree: number, t: number, knots: number[]): number {
    if (degree === 0) {
      return (knots[i] <= t && t < knots[i + 1]) ? 1 : 0;
    }
    
    const denom1 = knots[i + degree] - knots[i];
    const denom2 = knots[i + degree + 1] - knots[i + 1];
    
    let term1 = 0;
    let term2 = 0;
    
    if (denom1 !== 0) {
      term1 = ((t - knots[i]) / denom1) * this.basisFunction(i, degree - 1, t, knots);
    }
    
    if (denom2 !== 0) {
      term2 = ((knots[i + degree + 1] - t) / denom2) * this.basisFunction(i + 1, degree - 1, t, knots);
    }
    
    return term1 + term2;
  }

  getPoint(t: number): Point {
    const n = this.controlPoints.length - 1;
    let result: Point = [0, 0];
    
    for (let i = 0; i <= n; i++) {
      const basis = this.basisFunction(i, this.degree, t, this.knots!);
      result[0] += this.controlPoints[i][0] * basis;
      result[1] += this.controlPoints[i][1] * basis;
    }
    
    return result;
  }

  getPoints(segments: number = 100): Point[] {
    const points: Point[] = [];
    const tMin = this.knots![this.degree];
    const tMax = this.knots![this.controlPoints.length];
    
    for (let i = 0; i <= segments; i++) {
      const t = tMin + (tMax - tMin) * (i / segments);
      points.push(this.getPoint(t));
    }
    
    return points;
  }
}

// ========== NURBS曲线 (非均匀有理B样条) ==========
class NURBSCurve extends BSplineCurve {
  constructor(
    controlPoints: Point[],
    private weights: number[],
    degree: number = 3,
    knots?: number[]
  ) {
    super(controlPoints, degree, knots);
  }

  getPoint(t: number): Point {
    const n = this.controlPoints.length - 1;
    let numerator: Point = [0, 0];
    let denominator = 0;
    
    for (let i = 0; i <= n; i++) {
      const basis = this.basisFunction(i, this.degree, t, this.knots!);
      const weightedBasis = basis * this.weights[i];
      
      numerator[0] += this.controlPoints[i][0] * weightedBasis;
      numerator[1] += this.controlPoints[i][1] * weightedBasis;
      denominator += weightedBasis;
    }
    
    if (denominator === 0) return [0, 0];
    
    return [numerator[0] / denominator, numerator[1] / denominator];
  }
}

// ========== 埃尔米特曲线 ==========
class HermiteCurve extends Curve {
  constructor(
    private startPoint: Point,
    private endPoint: Point,
    private startTangent: Point,
    private endTangent: Point
  ) {
    super();
  }

  getPoint(t: number): Point {
    const t2 = t * t;
    const t3 = t2 * t;
    
    // 埃尔米特基函数
    const h1 = 2 * t3 - 3 * t2 + 1;
    const h2 = -2 * t3 + 3 * t2;
    const h3 = t3 - 2 * t2 + t;
    const h4 = t3 - t2;
    
    return [
      h1 * this.startPoint[0] + h2 * this.endPoint[0] + 
      h3 * this.startTangent[0] + h4 * this.endTangent[0],
      h1 * this.startPoint[1] + h2 * this.endPoint[1] + 
      h3 * this.startTangent[1] + h4 * this.endTangent[1]
    ];
  }

  getPoints(segments: number = 100): Point[] {
    const points: Point[] = [];
    for (let i = 0; i <= segments; i++) {
      const t = i / segments;
      points.push(this.getPoint(t));
    }
    return points;
  }
}

// ========== Catmull-Rom样条曲线 ==========
class CatmullRomCurve extends Curve {
  constructor(private points: Point[], private tension: number = 0.5) {
    super();
    if (points.length < 4) {
      throw new Error("Catmull-Rom曲线需要至少4个点");
    }
  }

  getPoint(t: number): Point {
    const n = this.points.length - 1;
    const segment = Math.floor(t * n);
    const localT = t * n - segment;
    
    if (segment < 1 || segment >= n - 1) {
      return this.points[Math.min(Math.max(segment, 0), n)];
    }
    
    const p0 = this.points[segment - 1];
    const p1 = this.points[segment];
    const p2 = this.points[segment + 1];
    const p3 = this.points[segment + 2];
    
    const t2 = localT * localT;
    const t3 = t2 * localT;
    
    return [
      0.5 * ((2 * p1[0]) + 
             (-p0[0] + p2[0]) * localT + 
             (2 * p0[0] - 5 * p1[0] + 4 * p2[0] - p3[0]) * t2 + 
             (-p0[0] + 3 * p1[0] - 3 * p2[0] + p3[0]) * t3),
      0.5 * ((2 * p1[1]) + 
             (-p0[1] + p2[1]) * localT + 
             (2 * p0[1] - 5 * p1[1] + 4 * p2[1] - p3[1]) * t2 + 
             (-p0[1] + 3 * p1[1] - 3 * p2[1] + p3[1]) * t3)
    ];
  }

  getPoints(segments: number = 100): Point[] {
    const points: Point[] = [];
    for (let i = 0; i <= segments; i++) {
      const t = i / segments;
      points.push(this.getPoint(t));
    }
    return points;
  }
}

// ========== 多项式曲线 ==========
class PolynomialCurve extends Curve {
  constructor(private coefficients: number[]) {
    super();
  }

  getPoint(t: number): Point {
    let x = 0;
    let y = 0;
    
    for (let i = 0; i < this.coefficients.length; i++) {
      const term = Math.pow(t, i);
      if (i % 2 === 0) {
        x += this.coefficients[i] * term;
      } else {
        y += this.coefficients[i] * term;
      }
    }
    
    return [x, y];
  }

  getPoints(segments: number = 100): Point[] {
    const points: Point[] = [];
    for (let i = 0; i <= segments; i++) {
      const t = -1 + 2 * (i / segments); // 在[-1, 1]范围内采样
      points.push(this.getPoint(t));
    }
    return points;
  }
}

// 曲线工厂类
class CurveFactory {
  static createCurve(type: string, points: Point[], options?: any): Curve {
    switch (type.toLowerCase()) {
      case 'bezier':
        return new BezierCurve(points);
      case 'bspline':
        return new BSplineCurve(points, options?.degree);
      case 'nurbs':
        return new NURBSCurve(points, options?.weights || points.map(() => 1), options?.degree);
      case 'hermite':
        return new HermiteCurve(points[0], points[1], options?.startTangent, options?.endTangent);
      case 'catmullrom':
        return new CatmullRomCurve(points, options?.tension);
      case 'polynomial':
        return new PolynomialCurve(options?.coefficients);
      default:
        throw new Error(`不支持的曲线类型: ${type}`);
    }
  }
}

// 使用工厂创建曲线
const factoryBezier = CurveFactory.createCurve('bezier', [[0, 0], [50, 100], [100, 50]]);
const factoryBSpline = CurveFactory.createCurve('bspline', [[0, 0], [30, 80], [70, 20]], { degree: 2 });
```

## 📈 曲线特性比较表

| 曲线类型 | 控制点影响 | 通过控制点 | 局部控制 | 计算复杂度 | 适用场景 |
|---------|-----------|-----------|----------|-----------|---------|
| **贝塞尔曲线** | 全局影响 | 首尾点 | 否 | 低 | UI设计、简单路径 |
| **B样条曲线** | 局部影响 | 一般不通过 | 是 | 中 | CAD、动画路径 |
| **NURBS曲线** | 局部影响+权重 | 一般不通过 | 是 | 高 | 工业设计、复杂曲面 |
| **埃尔米特曲线** | 端点+切线 | 首尾点 | 否 | 低 | 物理模拟、运动控制 |
| **Catmull-Rom** | 局部影响 | 所有点 | 是 | 中 | 相机路径、平滑插值 |
| **多项式曲线** | 系数控制 | 依赖系数 | 否 | 低 | 数学函数、数据拟合 |

## 🎯 选择指南

1. **简单UI动画**：选择贝塞尔曲线
2. **复杂路径设计**：选择B样条或Catmull-Rom曲线
3. **工业级建模**：选择NURBS曲线
4. **物理模拟**：选择埃尔米特曲线
5. **数学函数可视化**：选择多项式曲线

每种曲线都有其独特的数学特性和适用场景，在实际应用中可以根据具体需求选择合适的曲线类型。
