interface Point2D {
  x: number;
  y: number;

/**
 * 三种曲线类型的对比分析

export class CurveComparison {
  /**
   * 1. 贝塞尔曲线 (Bézier Curve)
   * 特点：基于伯恩斯坦多项式的参数曲线
   */
  export class BezierCurve {
    /**
     * 计算n阶贝塞尔曲线上的点
     */
    static evaluate(points: Point2D[], t: number): Point2D {
      const n = points.length - 1;
      const workingPoints = points.map(p => ({ ...p }));
      
      // 德卡斯特里奥算法
      for (let level = 1; level <= n; level++) {
        for (let i = 0; i <= n - level; i++) {
          workingPoints[i] = {
            x: (1 - t) * workingPoints[i].x + t * workingPoints[i + 1].x,
            y: (1 - t) * workingPoints[i].y + t * workingPoints[i + 1].y
          };
        }
      }
      
      return workingPoints[0];
    }
    
    /**
     * 生成贝塞尔曲线段
     */
    static generate(points: Point2D[], numSegments: number = 100): Point2D[] {
      const curve: Point2D[] = [];
      
      for (let i = 0; i <= numSegments; i++) {
        const t = i / numSegments;
        curve.push(this.evaluate(points, t));
      }
      
      return curve;
    }
    
    /**
     * 特点总结
     */
    static getCharacteristics(): string[] {
      return [
        "基于控制点，不通过所有控制点（端点除外）",
        "全局性：移动一个控制点影响整个曲线",
        "容易计算导数和曲率",
        "标准计算机图形学表示",
        "阶数 = 控制点数 - 1"
      ];
    }
  }
  
  /**
   * 2. Catmull-Rom 样条
   * 特点：通过所有控制点（关键点）的插值曲线
   */
  export class CatmullRomSpline {
    /**
     * 计算Catmull-Rom样条上的点
     * @param points 控制点（样条通过这些点）
     * @param t 参数 [0, 1]，在p1和p2之间
     * @param p0 前一个点
     * @param p1 起点
     * @param p2 终点
     * @param p3 后一个点
     * @param tension 张力参数（默认0.5）
     */
    static evaluate(
      p0: Point2D, p1: Point2D, p2: Point2D, p3: Point2D,
      t: number, tension: number = 0.5
    ): Point2D {
      // Catmull-Rom样条矩阵公式
      // 简化版：使用三次埃尔米特插值
      
      const t2 = t * t;
      const t3 = t2 * t;
      
      // 切线计算
      const m1 = {
        x: tension * (p2.x - p0.x),
        y: tension * (p2.y - p0.y)
      };
      
      const m2 = {
        x: tension * (p3.x - p1.x),
        y: tension * (p3.y - p1.y)
      };
      
      // 三次埃尔米特插值公式
      const h1 =  2 * t3 - 3 * t2 + 1;
      const h2 = -2 * t3 + 3 * t2;
      const h3 = t3 - 2 * t2 + t;
      const h4 = t3 - t2;
      
      return {
        x: h1 * p1.x + h2 * p2.x + h3 * m1.x + h4 * m2.x,
        y: h1 * p1.y + h2 * p2.y + h3 * m1.y + h4 * m2.y
      };
    }
    
    /**
     * 生成完整的Catmull-Rom样条
     */
    static generate(
      points: Point2D[], 
      closed: boolean = false,
      tension: number = 0.5,
      segmentsPerSegment: number = 20
    ): Point2D[] {
      if (points.length < 2) return [];
      
      const curve: Point2D[] = [];
      const n = points.length;
      
      // 处理闭合曲线
      const effectivePoints = closed 
        ? [points[n-1], ...points, points[0], points[1]]
        : points;
      
      // 起始点处理（非闭合）
      if (!closed) {
        curve.push(points[0]);
      }
      
      // 生成曲线段
      const start = closed ? 1 : 1;
      const end = closed ? effectivePoints.length - 2 : effectivePoints.length - 2;
      
      for (let i = start; i < end; i++) {
        const p0 = effectivePoints[i - 1];
        const p1 = effectivePoints[i];
        const p2 = effectivePoints[i + 1];
        const p3 = effectivePoints[i + 2];
        
        for (let j = 0; j <= segmentsPerSegment; j++) {
          const t = j / segmentsPerSegment;
          const point = this.evaluate(p0, p1, p2, p3, t, tension);
          
          // 避免重复添加端点
          if (j > 0 || i > start) {
            curve.push(point);
          }
        }
      }
      
      // 结束点处理（非闭合）
      if (!closed) {
        curve.push(points[n - 1]);
      }
      
      return curve;
    }
    
    /**
     * 特点总结
     */
    static getCharacteristics(): string[] {
      return [
        "通过所有控制点（插值曲线）",
        "局部性：移动一个点只影响相邻曲线段",
        "C1连续性（一阶导数连续）",
        "常用于动画和路径规划",
        "需要指定张力参数"
      ];
    }
  }
  
  /**
   * 3. 样条曲线 (Spline Curve)
   * 注：通常指B样条（B-spline）或自然样条
   */
  export class SplineCurve {
    /**
     * 三次B样条基函数
     */
    private static basisFunction(i: number, k: number, t: number, knots: number[]): number {
      // Cox-de Boor递归公式
      if (k === 1) {
        return (t >= knots[i] && t < knots[i + 1]) ? 1 : 0;
      }
      
      const denom1 = knots[i + k - 1] - knots[i];
      const denom2 = knots[i + k] - knots[i + 1];
      
      let term1 = 0;
      let term2 = 0;
      
      if (denom1 !== 0) {
        term1 = ((t - knots[i]) / denom1) * this.basisFunction(i, k - 1, t, knots);
      }
      
      if (denom2 !== 0) {
        term2 = ((knots[i + k] - t) / denom2) * this.basisFunction(i + 1, k - 1, t, knots);
      }
      
      return term1 + term2;
    }
    
    /**
     * 均匀B样条
     */
    static evaluateUniformBSpline(
      points: Point2D[], 
      degree: number, 
      t: number
    ): Point2D {
      const n = points.length - 1;
      const m = n + degree + 1;
      
      // 均匀节点向量
      const knots: number[] = [];
      for (let i = 0; i <= m; i++) {
        knots.push(i);
      }
      
      const tNormalized = t * (n - degree + 1);
      
      let result = { x: 0, y: 0 };
      
      for (let i = 0; i <= n; i++) {
        const basis = this.basisFunction(i, degree + 1, tNormalized, knots);
        result.x += points[i].x * basis;
        result.y += points[i].y * basis;
      }
      
      return result;
    }
    
    /**
     * 自然三次样条（C2连续）
     */
    static evaluateNaturalCubicSpline(
      points: Point2D[],
      t: number,
      segmentIndex: number
    ): Point2D {
      // 简化实现：使用分段三次多项式
      // 实际实现需要求解三对角方程组
      
      // 这里使用Catmull-Rom作为近似
      return { x: 0, y: 0 };
    }
    
    /**
     * 生成均匀B样条曲线
     */
    static generateUniformBSpline(
      points: Point2D[],
      degree: number = 3,
      numSegments: number = 100
    ): Point2D[] {
      const curve: Point2D[] = [];
      
      for (let i = 0; i <= numSegments; i++) {
        const t = i / numSegments;
        curve.push(this.evaluateUniformBSpline(points, degree, t));
      }
      
      return curve;
    }
    
    /**
     * 特点总结
     */
    static getCharacteristics(): string[] {
      return [
        "B样条：不通过控制点（除非重复节点）",
        "自然样条：通过所有点且C2连续",
        "局部支撑性：修改一个控制点只影响局部",
        "可以表示复杂形状而无需高次多项式",
        "节点向量控制曲线特性"
      ];
    }
  }
}

export class CurveAnalyzer {
  /**
   * 比较三种曲线在相同控制点下的表现
   */
  static compareCurves(
    controlPoints: Point2D[],
    options: {
      catmullRomTension?: number;
      bSplineDegree?: number;
      segments?: number;
    } = {}
  ): {
    bezier: Point2D[];
    catmullRom: Point2D[];
    bSpline: Point2D[];
    analysis: Record<string, any>;
  } {
    const tension = options.catmullRomTension || 0.5;
    const degree = options.bSplineDegree || 3;
    const segments = options.segments || 100;
    
    // 生成各种曲线
    const bezierCurve = CurveComparison.BezierCurve.generate(
      controlPoints, segments
    );
    
    const catmullRomCurve = CurveComparison.CatmullRomSpline.generate(
      controlPoints, false, tension, Math.floor(segments / controlPoints.length)
    );
    
    const bSplineCurve = CurveComparison.SplineCurve.generateUniformBSpline(
      controlPoints, degree, segments
    );
    
    // 分析曲线特性
    const analysis = {
      controlPointsCount: controlPoints.length,
      bezierDegree: controlPoints.length - 1,
      continuity: {
        bezier: "C∞ (无限可微，但高次可能振荡)",
        catmullRom: "C1 (一阶导数连续)",
        bSpline: `C${degree-1} (${degree-1}阶导数连续)`
      },
      interpolation: {
        bezier: "只通过起点和终点",
        catmullRom: "通过所有控制点",
        bSpline: "一般不通过控制点（近似）"
      },
      locality: {
        bezier: "全局影响",
        catmullRom: "局部影响（相邻4个点）",
        bSpline: "局部影响（相邻degree+1个点）"
      }
    };
    
    return {
      bezier: bezierCurve,
      catmullRom: catmullRomCurve,
      bSpline: bSplineCurve,
      analysis
    };
  }
  
  /**
   * 计算曲线长度（数值积分）
   */
  static curveLength(curve: Point2D[]): number {
    let length = 0;
    
    for (let i = 1; i < curve.length; i++) {
      const dx = curve[i].x - curve[i - 1].x;
      const dy = curve[i].y - curve[i - 1].y;
      length += Math.sqrt(dx * dx + dy * dy);
    }
    
    return length;
  }
  
  /**
   * 计算曲线平滑度（基于曲率变化）
   */
  static curveSmoothness(curve: Point2D[]): number {
    // 简单的平滑度度量：角度变化总和
    let totalAngleChange = 0;
    
    for (let i = 1; i < curve.length - 1; i++) {
      const v1 = {
        x: curve[i].x - curve[i - 1].x,
        y: curve[i].y - curve[i - 1].y
      };
      
      const v2 = {
        x: curve[i + 1].x - curve[i].x,
        y: curve[i + 1].y - curve[i].y
      };
      
      const dot = v1.x * v2.x + v1.y * v2.y;
      const mag1 = Math.sqrt(v1.x * v1.x + v1.y * v1.y);
      const mag2 = Math.sqrt(v2.x * v2.x + v2.y * v2.y);
      
      if (mag1 > 0 && mag2 > 0) {
        const cosAngle = Math.max(-1, Math.min(1, dot / (mag1 * mag2)));
        const angle = Math.acos(cosAngle);
        totalAngleChange += angle;
      }
    }
    
    // 平滑度：角度变化越小越平滑
    return 1 / (1 + totalAngleChange);
  }
}

## 三种曲线的核心区别总结

### 1. **数学定义**

| 特性 | 贝塞尔曲线 | Catmull-Rom样条 | B样条 |
|------|-----------|----------------|-------|
| **数学基础** | 伯恩斯坦多项式 | 分段三次埃尔米特插值 | 基函数线性组合 |
| **公式** | $B(t) = \sum B_{i,n}(t)P_i$ | 分段公式，使用相邻4点 | $S(t) = \sum N_{i,k}(t)P_i$ |
| **连续性** | $C^\infty$ | $C^1$ | $C^{k-2}$ |

### 2. **几何特性**

#### **贝塞尔曲线**：
- **凸包性**：曲线完全在控制点凸包内
- **端点插值**：通过第一个和最后一个控制点
- **端点切线**：与第一条和最后一条控制多边形边相切
- **全局性**：修改任意控制点影响整个曲线

#### **Catmull-Rom样条**：
- **插值性**：通过所有控制点
- **局部性**：只依赖相邻4个控制点
- **C1连续**：一阶导数连续
- **张力参数**：可调整曲线松紧

#### **B样条**：
- **局部支撑**：每个基函数只影响局部区间
- **强凸包**：曲线在k+1个连续控制点的凸包内
- **变差缩减**：比控制多边形更平滑
- **节点向量**：控制曲线参数化

### 3. **控制点关系**

```typescript
// 三种曲线对控制点的不同处理方式

const points = [P0, P1, P2, P3, P4, P5];

// 贝塞尔曲线：使用所有点定义一个高阶曲线
// B(t) 使用 P0-P5 所有点

// Catmull-Rom：分段处理，每段使用4个点
// 段1: P0,P1,P2,P3
// 段2: P1,P2,P3,P4
// 段3: P2,P3,P4,P5

// B样条：每个点对基函数有贡献，贡献范围有限
// S(t) = N0,k(t)P0 + N1,k(t)P1 + ... + N5,k(t)P5
```

### 7. **性能建议**

1. **贝塞尔曲线**：
   - 当控制点 < 10 时使用
   - 避免高阶贝塞尔曲线（>5阶）
   - 使用德卡斯特里奥算法求值

2. **Catmull-Rom样条**：
   - 需要插值大量点时使用
   - 调整张力参数优化曲线形状
   - 注意边界条件处理

3. **B样条**：
   - 复杂形状时使用
   - 预计算基函数提高性能
   - 选择合适的节点向量

### 8. **实际选择建议**

```typescript
function selectCurveType(requirements: {
  mustInterpolate: boolean;
  numPoints: number;
  continuity: number;
  localControl: boolean;
  implementationComplexity: "low" | "medium" | "high";
}): string {
  if (requirements.mustInterpolate && requirements.numPoints > 2) {
    return "Catmull-Rom";
  }
  
  if (requirements.numPoints <= 4 && !requirements.localControl) {
    return "Bezier";
  }
  
  if (requirements.continuity >= 2 || requirements.numPoints > 10) {
    return "B-spline";
  }
  
  return "Bezier"; // 默认选择
}
```

三种曲线各有优劣，贝塞尔曲线简单直观，Catmull-Rom适合插值需求，B样条数学性质最好但实现复杂。实际应用中常根据具体需求组合使用。
